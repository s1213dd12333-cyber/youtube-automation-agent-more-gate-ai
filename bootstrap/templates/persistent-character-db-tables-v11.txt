      `CREATE TABLE IF NOT EXISTS persistent_characters (
        id TEXT PRIMARY KEY,
        namespace TEXT NOT NULL DEFAULT 'default',
        character_key TEXT,
        display_name TEXT NOT NULL,
        species_type TEXT NOT NULL DEFAULT 'character',
        version TEXT NOT NULL DEFAULT '11.11.1',
        identity_fingerprint TEXT NOT NULL,
        canonical_identity TEXT NOT NULL DEFAULT '{}',
        canonical_prompt_context TEXT NOT NULL DEFAULT '',
        status TEXT NOT NULL DEFAULT 'active',
        created_from_production_id TEXT,
        first_seen_at TEXT NOT NULL,
        last_seen_at TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(namespace, identity_fingerprint),
        UNIQUE(namespace, character_key)
      )`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_characters_namespace ON persistent_characters(namespace, status, updated_at)`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_characters_name ON persistent_characters(namespace, display_name)`,
      `CREATE TABLE IF NOT EXISTS persistent_character_usages (
        id TEXT PRIMARY KEY,
        character_id TEXT NOT NULL,
        production_id TEXT NOT NULL,
        source_bible_id TEXT,
        source_character_id TEXT,
        role TEXT,
        source_kind TEXT,
        source_ref TEXT,
        match_mode TEXT NOT NULL,
        identity_fingerprint TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(character_id, production_id, source_character_id),
        FOREIGN KEY (character_id) REFERENCES persistent_characters(id) ON DELETE CASCADE,
        FOREIGN KEY (production_id) REFERENCES productions(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_character_usages_production ON persistent_character_usages(production_id, created_at)`,
