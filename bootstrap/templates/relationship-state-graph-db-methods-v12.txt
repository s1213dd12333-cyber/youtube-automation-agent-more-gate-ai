  serializedRelationshipSnapshot(relationship = {}) {
    const stable = value => {
      if (Array.isArray(value)) return value.map(stable);
      if (!value || typeof value !== 'object') return value;
      return Object.keys(value).sort().reduce((out, key) => { out[key] = stable(value[key]); return out; }, {});
    };
    return stable({
      id: relationship.id,
      seriesId: relationship.seriesId,
      sourceCharacterKey: relationship.sourceCharacterKey,
      targetCharacterKey: relationship.targetCharacterKey,
      sourceCharacterArcId: relationship.sourceCharacterArcId || null,
      targetCharacterArcId: relationship.targetCharacterArcId || null,
      relationshipLabel: relationship.relationshipLabel || '',
      relationshipState: relationship.relationshipState || '',
      relationshipTags: relationship.relationshipTags || [],
      trustScore: Number(relationship.trustScore || 0),
      affinityScore: Number(relationship.affinityScore || 0),
      respectScore: Number(relationship.respectScore || 0),
      loyaltyScore: Number(relationship.loyaltyScore || 0),
      fearScore: Number(relationship.fearScore || 0),
      attractionScore: Number(relationship.attractionScore || 0),
      dependenceScore: Number(relationship.dependenceScore || 0),
      beliefsAboutTarget: relationship.beliefsAboutTarget || [],
      knowledgeAboutTarget: relationship.knowledgeAboutTarget || [],
      secretsKnownAboutTarget: relationship.secretsKnownAboutTarget || [],
      obligationsToTarget: relationship.obligationsToTarget || [],
      promisesToTarget: relationship.promisesToTarget || [],
      grievancesAgainstTarget: relationship.grievancesAgainstTarget || [],
      expectationsOfTarget: relationship.expectationsOfTarget || [],
      boundariesWithTarget: relationship.boundariesWithTarget || [],
      sharedHistory: relationship.sharedHistory || [],
      currentTensions: relationship.currentTensions || [],
      notes: relationship.notes || '',
      firstEpisodeNumber: Number(relationship.firstEpisodeNumber || 0),
      lastEpisodeNumber: Number(relationship.lastEpisodeNumber || 0),
      revisionNumber: Number(relationship.revisionNumber || 0),
      status: relationship.status || 'active'
    });
  }

  async commitSerializedRelationshipsAtomic(input = {}) {
    if (!input.seriesId || !input.episodeNumber || !input.episodeMemoryId) throw new Error('relationship_commit_identity_required');
    if (!Array.isArray(input.updates) || !input.updates.length) throw new Error('relationship_updates_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const series = await this.getRow('SELECT * FROM serialized_series_bibles WHERE id = ? LIMIT 1', [input.seriesId]);
      if (!series) throw new Error('relationship_series_not_found');
      if ((series.status || 'active') !== 'active') throw new Error('relationship_series_inactive');
      const episodeNumber = Number(input.episodeNumber);
      const currentEpisode = Number(series.current_episode || 0);
      if (episodeNumber > currentEpisode) throw new Error('relationship_future_episode_not_finalized');
      if (episodeNumber < currentEpisode && input.allowBackfill !== true) throw new Error('relationship_backfill_requires_explicit_approval');
      const memoryRow = await this.getRow(
        `SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? AND episode_number = ? AND status = 'finalized' LIMIT 1`,
        [input.episodeMemoryId, input.seriesId, episodeNumber]
      );
      if (!memoryRow) throw new Error('finalized_episode_memory_required');
      let memoryTimeline = []; try { memoryTimeline = JSON.parse(memoryRow.timeline_event_ids || '[]'); } catch (_error) {}
      const allowedTimeline = new Set(memoryTimeline);
      const now = new Date().toISOString();
      const relationshipIds = [];
      const commitIds = [];

      for (const item of input.updates) {
        const relationship = item.relationship || {};
        if (!relationship.sourceCharacterKey || !relationship.targetCharacterKey) throw new Error('relationship_endpoints_required');
        if (relationship.sourceCharacterKey === relationship.targetCharacterKey) throw new Error('relationship_self_edge_forbidden');
        const sourceArc = await this.getRow(
          'SELECT * FROM serialized_character_arcs WHERE id = ? AND series_id = ? AND character_key = ? LIMIT 1',
          [relationship.sourceCharacterArcId, input.seriesId, relationship.sourceCharacterKey]
        );
        const targetArc = await this.getRow(
          'SELECT * FROM serialized_character_arcs WHERE id = ? AND series_id = ? AND character_key = ? LIMIT 1',
          [relationship.targetCharacterArcId, input.seriesId, relationship.targetCharacterKey]
        );
        if (!sourceArc) throw new Error('relationship_source_character_arc_not_found');
        if (!targetArc) throw new Error('relationship_target_character_arc_not_found');

        const row = await this.getRow(
          `SELECT * FROM serialized_relationship_states
           WHERE series_id = ? AND source_character_key = ? AND target_character_key = ? LIMIT 1`,
          [input.seriesId, relationship.sourceCharacterKey, relationship.targetCharacterKey]
        );
        if (row && (row.source_character_arc_id !== relationship.sourceCharacterArcId || row.target_character_arc_id !== relationship.targetCharacterArcId)) {
          throw new Error('relationship_endpoint_arc_binding_drift');
        }
        const currentRevision = Number(row?.revision_number || 0);
        if (Number(item.expectedRevision || 0) !== currentRevision) throw new Error('relationship_revision_conflict');
        const relationshipId = row?.id || relationship.id || this.generateId('serialized_relationship');
        const latest = row ? await this.getRow(
          `SELECT * FROM serialized_relationship_episode_commits WHERE relationship_id = ? ORDER BY relationship_revision_number DESC LIMIT 1`,
          [relationshipId]
        ) : null;
        if (latest && Number(latest.episode_number || 0) > episodeNumber) throw new Error('relationship_historical_insert_has_later_commit');
        const duplicate = row ? await this.getRow(
          `SELECT id FROM serialized_relationship_episode_commits
           WHERE relationship_id = ? AND episode_number = ? AND commit_kind = 'episode_commit' LIMIT 1`,
          [relationshipId, episodeNumber]
        ) : null;
        if (duplicate) throw new Error('relationship_episode_already_committed');
        const timelineIds = [...new Set(item.relevantTimelineEventIds || [])];
        if (timelineIds.some(id => !allowedTimeline.has(id))) throw new Error('relationship_timeline_reference_outside_episode_memory');

        const before = row ? this.serializedRelationshipSnapshot(this.parseSerializedRelationshipState(row)) : {};
        const nextRevision = currentRevision + 1;
        const firstEpisode = row ? Number(row.first_episode_number || episodeNumber) : episodeNumber;
        const values = [
          relationship.relationshipLabel || '', relationship.relationshipState || '', JSON.stringify(relationship.relationshipTags || []),
          Number(relationship.trustScore || 0), Number(relationship.affinityScore || 0), Number(relationship.respectScore || 0),
          Number(relationship.loyaltyScore || 0), Number(relationship.fearScore || 0), Number(relationship.attractionScore || 0),
          Number(relationship.dependenceScore || 0), JSON.stringify(relationship.beliefsAboutTarget || []),
          JSON.stringify(relationship.knowledgeAboutTarget || []), JSON.stringify(relationship.secretsKnownAboutTarget || []),
          JSON.stringify(relationship.obligationsToTarget || []), JSON.stringify(relationship.promisesToTarget || []),
          JSON.stringify(relationship.grievancesAgainstTarget || []), JSON.stringify(relationship.expectationsOfTarget || []),
          JSON.stringify(relationship.boundariesWithTarget || []), JSON.stringify(relationship.sharedHistory || []),
          JSON.stringify(relationship.currentTensions || []), relationship.notes || ''
        ];
        if (!row) {
          await this.executeQuery(
            `INSERT INTO serialized_relationship_states (
              id, series_id, source_character_key, target_character_key, source_character_arc_id, target_character_arc_id,
              relationship_label, relationship_state, relationship_tags, trust_score, affinity_score, respect_score, loyalty_score,
              fear_score, attraction_score, dependence_score, beliefs_about_target, knowledge_about_target, secrets_known_about_target,
              obligations_to_target, promises_to_target, grievances_against_target, expectations_of_target, boundaries_with_target,
              shared_history, current_tensions, notes, first_episode_number, last_episode_number, revision_number, status, created_at, updated_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
            [relationshipId, input.seriesId, relationship.sourceCharacterKey, relationship.targetCharacterKey,
              relationship.sourceCharacterArcId, relationship.targetCharacterArcId, ...values, firstEpisode, episodeNumber,
              nextRevision, relationship.status || 'active', now, now]
          );
        } else {
          await this.executeQuery(
            `UPDATE serialized_relationship_states SET
              relationship_label = ?, relationship_state = ?, relationship_tags = ?, trust_score = ?, affinity_score = ?,
              respect_score = ?, loyalty_score = ?, fear_score = ?, attraction_score = ?, dependence_score = ?, beliefs_about_target = ?,
              knowledge_about_target = ?, secrets_known_about_target = ?, obligations_to_target = ?, promises_to_target = ?,
              grievances_against_target = ?, expectations_of_target = ?, boundaries_with_target = ?, shared_history = ?,
              current_tensions = ?, notes = ?, last_episode_number = ?, revision_number = ?, status = ?, updated_at = ?
             WHERE id = ? AND series_id = ? AND revision_number = ?`,
            [...values, episodeNumber, nextRevision, relationship.status || 'active', now, relationshipId, input.seriesId, currentRevision]
          );
          const check = await this.getRow('SELECT revision_number FROM serialized_relationship_states WHERE id = ? LIMIT 1', [relationshipId]);
          if (Number(check?.revision_number || -1) !== nextRevision) throw new Error('relationship_revision_conflict');
        }
        const saved = await this.getSerializedRelationshipStateById(relationshipId);
        const after = this.serializedRelationshipSnapshot(saved);
        await this.executeQuery(
          `INSERT INTO serialized_relationship_revisions (
            id, relationship_id, series_id, episode_number, revision_number, snapshot, change_kind, change_reason, actor, created_at
          ) VALUES (?, ?, ?, ?, ?, ?, 'episode_commit', ?, ?, ?)`,
          [this.generateId('serialized_relationship_revision'), relationshipId, input.seriesId, episodeNumber, nextRevision,
            JSON.stringify(after), input.reason || '', input.actor || null, now]
        );
        const commitId = this.generateId('serialized_relationship_commit');
        await this.executeQuery(
          `INSERT INTO serialized_relationship_episode_commits (
            id, series_id, relationship_id, episode_memory_id, episode_number, relationship_revision_number, commit_kind,
            timeline_event_ids, before_snapshot, after_snapshot, change_summary, reason, actor, created_at
          ) VALUES (?, ?, ?, ?, ?, ?, 'episode_commit', ?, ?, ?, ?, ?, ?, ?)`,
          [commitId, input.seriesId, relationshipId, input.episodeMemoryId, episodeNumber, nextRevision,
            JSON.stringify(timelineIds), JSON.stringify(before), JSON.stringify(after), item.changeSummary || '',
            input.reason || '', input.actor || null, now]
        );
        relationshipIds.push(relationshipId);
        commitIds.push(commitId);
      }
      await this.executeQuery('COMMIT');
      const relationships = [];
      for (const id of relationshipIds) relationships.push(await this.getSerializedRelationshipStateById(id));
      const commits = [];
      for (const id of commitIds) commits.push(this.parseSerializedRelationshipCommit(await this.getRow(
        'SELECT * FROM serialized_relationship_episode_commits WHERE id = ? LIMIT 1', [id]
      )));
      return { relationships, commits };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }

  async amendSerializedRelationshipAtomic(input = {}) {
    if (!input.seriesId || !input.relationship?.id || !input.episodeMemoryId) throw new Error('relationship_amendment_identity_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const row = await this.getRow(
        'SELECT * FROM serialized_relationship_states WHERE id = ? AND series_id = ? LIMIT 1',
        [input.relationship.id, input.seriesId]
      );
      if (!row) throw new Error('relationship_not_found');
      const currentRevision = Number(row.revision_number || 0);
      if (Number(input.expectedRevision) !== currentRevision) throw new Error('relationship_revision_conflict');
      if (row.source_character_key !== input.relationship.sourceCharacterKey || row.target_character_key !== input.relationship.targetCharacterKey) {
        throw new Error('relationship_endpoints_are_immutable');
      }
      if (row.source_character_arc_id !== input.relationship.sourceCharacterArcId || row.target_character_arc_id !== input.relationship.targetCharacterArcId) {
        throw new Error('relationship_endpoint_arc_binding_drift');
      }
      const sourceArc = await this.getRow(
        'SELECT id FROM serialized_character_arcs WHERE id = ? AND series_id = ? AND character_key = ? LIMIT 1',
        [input.relationship.sourceCharacterArcId, input.seriesId, input.relationship.sourceCharacterKey]
      );
      const targetArc = await this.getRow(
        'SELECT id FROM serialized_character_arcs WHERE id = ? AND series_id = ? AND character_key = ? LIMIT 1',
        [input.relationship.targetCharacterArcId, input.seriesId, input.relationship.targetCharacterKey]
      );
      if (!sourceArc) throw new Error('relationship_source_character_arc_not_found');
      if (!targetArc) throw new Error('relationship_target_character_arc_not_found');
      const latest = await this.getRow(
        `SELECT * FROM serialized_relationship_episode_commits WHERE relationship_id = ? ORDER BY relationship_revision_number DESC LIMIT 1`,
        [input.relationship.id]
      );
      if (!latest || Number(latest.episode_number || 0) !== Number(input.episodeNumber)) throw new Error('historical_relationship_amendment_blocked_by_later_state');
      const memory = await this.getRow(
        `SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? AND episode_number = ? AND status = 'finalized' LIMIT 1`,
        [input.episodeMemoryId, input.seriesId, Number(input.episodeNumber)]
      );
      if (!memory) throw new Error('finalized_episode_memory_required');
      let allowedIds = []; try { allowedIds = JSON.parse(memory.timeline_event_ids || '[]'); } catch (_error) {}
      const allowed = new Set(allowedIds);
      const timelineIds = [...new Set(input.relevantTimelineEventIds || [])];
      if (timelineIds.some(id => !allowed.has(id))) throw new Error('relationship_timeline_reference_outside_episode_memory');

      const before = this.serializedRelationshipSnapshot(this.parseSerializedRelationshipState(row));
      const nextRevision = currentRevision + 1;
      const now = new Date().toISOString();
      const relationship = input.relationship;
      await this.executeQuery(
        `UPDATE serialized_relationship_states SET
          relationship_label = ?, relationship_state = ?, relationship_tags = ?, trust_score = ?, affinity_score = ?, respect_score = ?,
          loyalty_score = ?, fear_score = ?, attraction_score = ?, dependence_score = ?, beliefs_about_target = ?, knowledge_about_target = ?,
          secrets_known_about_target = ?, obligations_to_target = ?, promises_to_target = ?, grievances_against_target = ?,
          expectations_of_target = ?, boundaries_with_target = ?, shared_history = ?, current_tensions = ?, notes = ?, revision_number = ?,
          status = ?, updated_at = ? WHERE id = ? AND series_id = ? AND revision_number = ?`,
        [relationship.relationshipLabel || '', relationship.relationshipState || '', JSON.stringify(relationship.relationshipTags || []),
          Number(relationship.trustScore || 0), Number(relationship.affinityScore || 0), Number(relationship.respectScore || 0),
          Number(relationship.loyaltyScore || 0), Number(relationship.fearScore || 0), Number(relationship.attractionScore || 0),
          Number(relationship.dependenceScore || 0), JSON.stringify(relationship.beliefsAboutTarget || []),
          JSON.stringify(relationship.knowledgeAboutTarget || []), JSON.stringify(relationship.secretsKnownAboutTarget || []),
          JSON.stringify(relationship.obligationsToTarget || []), JSON.stringify(relationship.promisesToTarget || []),
          JSON.stringify(relationship.grievancesAgainstTarget || []), JSON.stringify(relationship.expectationsOfTarget || []),
          JSON.stringify(relationship.boundariesWithTarget || []), JSON.stringify(relationship.sharedHistory || []),
          JSON.stringify(relationship.currentTensions || []), relationship.notes || '', nextRevision,
          relationship.status || 'active', now, relationship.id, input.seriesId, currentRevision]
      );
      const saved = await this.getSerializedRelationshipStateById(relationship.id);
      if (Number(saved?.revisionNumber || -1) !== nextRevision) throw new Error('relationship_revision_conflict');
      const after = this.serializedRelationshipSnapshot(saved);
      await this.executeQuery(
        `INSERT INTO serialized_relationship_revisions (
          id, relationship_id, series_id, episode_number, revision_number, snapshot, change_kind, change_reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?)`,
        [this.generateId('serialized_relationship_revision'), relationship.id, input.seriesId, Number(input.episodeNumber),
          nextRevision, JSON.stringify(after), input.reason || '', input.actor || null, now]
      );
      const commitId = this.generateId('serialized_relationship_commit');
      await this.executeQuery(
        `INSERT INTO serialized_relationship_episode_commits (
          id, series_id, relationship_id, episode_memory_id, episode_number, relationship_revision_number, commit_kind,
          timeline_event_ids, before_snapshot, after_snapshot, change_summary, reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?, ?, ?, ?, ?)`,
        [commitId, input.seriesId, relationship.id, input.episodeMemoryId, Number(input.episodeNumber), nextRevision,
          JSON.stringify(timelineIds), JSON.stringify(before), JSON.stringify(after), input.changeSummary || '', input.reason || '',
          input.actor || null, now]
      );
      await this.executeQuery('COMMIT');
      return {
        relationship: await this.getSerializedRelationshipStateById(relationship.id),
        commit: this.parseSerializedRelationshipCommit(await this.getRow(
          'SELECT * FROM serialized_relationship_episode_commits WHERE id = ? LIMIT 1', [commitId]
        ))
      };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }
