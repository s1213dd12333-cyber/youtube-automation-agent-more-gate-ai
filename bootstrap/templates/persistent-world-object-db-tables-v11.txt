      // Phase 11.10.1 production-independent Persistent World Object Registry
      `CREATE TABLE IF NOT EXISTS persistent_world_objects (
        id TEXT PRIMARY KEY,
        namespace TEXT NOT NULL DEFAULT 'default',
        object_key TEXT NOT NULL,
        display_name TEXT NOT NULL,
        object_type TEXT NOT NULL DEFAULT 'object',
        version TEXT NOT NULL DEFAULT '11.10.1',
        identity_fingerprint TEXT NOT NULL,
        canonical_identity TEXT NOT NULL DEFAULT '{}',
        canonical_prompt_context TEXT NOT NULL DEFAULT '',
        canonical_location_id TEXT,
        canonical_zone_id TEXT,
        canonical_prop_lock_id TEXT,
        status TEXT NOT NULL DEFAULT 'registered',
        created_from_production_id TEXT,
        first_seen_at TEXT NOT NULL,
        last_seen_at TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(namespace, identity_fingerprint),
        UNIQUE(namespace, object_key),
        FOREIGN KEY (canonical_location_id) REFERENCES reusable_locations(id) ON DELETE SET NULL,
        FOREIGN KEY (canonical_zone_id) REFERENCES reusable_location_zones(id) ON DELETE SET NULL
      )`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_world_objects_namespace ON persistent_world_objects(namespace, status, last_seen_at)`,
      `CREATE TABLE IF NOT EXISTS persistent_world_object_usages (
        id TEXT PRIMARY KEY,
        object_id TEXT NOT NULL,
        production_id TEXT NOT NULL,
        scope_key TEXT NOT NULL,
        environment_id TEXT,
        scene_id TEXT,
        location_id TEXT,
        zone_id TEXT,
        prop_lock_id TEXT,
        source_kind TEXT NOT NULL DEFAULT 'explicit_world_object',
        source_ref TEXT,
        match_mode TEXT NOT NULL DEFAULT 'identity_fingerprint_exact_register',
        identity_fingerprint TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(object_id, production_id, scope_key),
        FOREIGN KEY (object_id) REFERENCES persistent_world_objects(id) ON DELETE CASCADE,
        FOREIGN KEY (production_id) REFERENCES productions(id) ON DELETE CASCADE,
        FOREIGN KEY (location_id) REFERENCES reusable_locations(id) ON DELETE SET NULL,
        FOREIGN KEY (zone_id) REFERENCES reusable_location_zones(id) ON DELETE SET NULL
      )`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_world_object_usages_prod ON persistent_world_object_usages(production_id, object_id, created_at)`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_world_object_usages_object ON persistent_world_object_usages(object_id, production_id, created_at)`,
