  async savePersistentCharacterAsset(input = {}) {
    if (!input.assetKey || !input.characterId || !input.identityFingerprint || !input.assetPath || !input.assetSha256) return null;
    const existing = await this.getPersistentCharacterAssetByKey(input.assetKey);
    const id = existing?.id || input.id || this.generateId('persistent_character_asset');
    const now = new Date().toISOString();
    const createdAt = existing?.createdAt || input.createdAt || now;
    await this.executeQuery(`INSERT INTO persistent_character_assets (
      id, asset_key, character_id, asset_role, version, identity_fingerprint, canonical, status, asset_path, asset_sha256,
      source_kind, source_production_id, source_bible_id, source_character_id, source_ref, provider, model, created_at, updated_at
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    ON CONFLICT(asset_key) DO UPDATE SET updated_at=excluded.updated_at`, [
      id, input.assetKey, input.characterId, input.assetRole || 'character_reference', String(input.version || '11.11.3'), input.identityFingerprint,
      input.canonical === false ? 0 : 1, input.status || 'ready', input.assetPath, input.assetSha256, input.sourceKind || 'explicit-canonical-character-reference',
      input.sourceProductionId || null, input.sourceBibleId || null, input.sourceCharacterId || null, input.sourceRef || null,
      input.provider || null, input.model || null, createdAt, now
    ]);
    return this.getPersistentCharacterAssetByKey(input.assetKey);
  }

  async getPersistentCharacterAsset(id) {
    return this.parsePersistentCharacterAsset(await this.getRow('SELECT * FROM persistent_character_assets WHERE id = ? LIMIT 1', [id]));
  }

  async getPersistentCharacterAssetByKey(assetKey) {
    return this.parsePersistentCharacterAsset(await this.getRow('SELECT * FROM persistent_character_assets WHERE asset_key = ? LIMIT 1', [assetKey]));
  }

  async getPersistentCharacterAssetByCharacter(characterId, assetRole = 'character_reference') {
    return this.parsePersistentCharacterAsset(await this.getRow('SELECT * FROM persistent_character_assets WHERE character_id = ? AND asset_role = ? LIMIT 1', [characterId, assetRole]));
  }

  async listPersistentCharacterAssets(characterId = null) {
    const rows = characterId
      ? await this.getAllRows('SELECT * FROM persistent_character_assets WHERE character_id = ? ORDER BY created_at, rowid', [characterId])
      : await this.getAllRows('SELECT * FROM persistent_character_assets ORDER BY created_at, rowid', []);
    return rows.map(row => this.parsePersistentCharacterAsset(row));
  }

  async listProductionPersistentCharacterAssets(productionId) {
    const rows = await this.getAllRows(`SELECT DISTINCT a.* FROM persistent_character_assets a
      JOIN persistent_character_usages u ON u.character_id = a.character_id
      WHERE u.production_id = ? ORDER BY a.created_at, a.rowid`, [productionId]);
    return rows.map(row => this.parsePersistentCharacterAsset(row));
  }

  parsePersistentCharacterAsset(row) {
    if (!row) return null;
    return {
      id: row.id, assetKey: row.asset_key, characterId: row.character_id, assetRole: row.asset_role || 'character_reference', version: row.version || '11.11.3',
      identityFingerprint: row.identity_fingerprint, canonical: row.canonical === 1 || row.canonical === true, status: row.status || 'ready',
      assetPath: row.asset_path, assetSha256: row.asset_sha256, sourceKind: row.source_kind, sourceProductionId: row.source_production_id,
      sourceBibleId: row.source_bible_id, sourceCharacterId: row.source_character_id, sourceRef: row.source_ref,
      provider: row.provider, model: row.model, createdAt: row.created_at, updatedAt: row.updated_at
    };
  }
