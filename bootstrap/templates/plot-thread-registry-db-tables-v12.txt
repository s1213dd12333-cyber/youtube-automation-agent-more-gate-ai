      `CREATE TABLE IF NOT EXISTS serialized_plot_threads (
        id TEXT PRIMARY KEY,
        series_id TEXT NOT NULL,
        thread_key TEXT NOT NULL,
        title TEXT NOT NULL,
        thread_type TEXT NOT NULL DEFAULT 'other',
        priority INTEGER NOT NULL DEFAULT 50,
        premise TEXT NOT NULL DEFAULT '',
        central_question TEXT NOT NULL DEFAULT '',
        stakes TEXT NOT NULL DEFAULT '',
        current_state TEXT NOT NULL DEFAULT '',
        open_questions TEXT NOT NULL DEFAULT '[]',
        resolved_questions TEXT NOT NULL DEFAULT '[]',
        narrative_promises TEXT NOT NULL DEFAULT '[]',
        established_clues TEXT NOT NULL DEFAULT '[]',
        red_herrings TEXT NOT NULL DEFAULT '[]',
        required_payoffs TEXT NOT NULL DEFAULT '[]',
        involved_character_keys TEXT NOT NULL DEFAULT '[]',
        relationship_edge_keys TEXT NOT NULL DEFAULT '[]',
        resolution_summary TEXT NOT NULL DEFAULT '',
        notes TEXT NOT NULL DEFAULT '',
        introduced_episode_number INTEGER NOT NULL DEFAULT 0,
        last_advanced_episode_number INTEGER NOT NULL DEFAULT 0,
        resolved_episode_number INTEGER NOT NULL DEFAULT 0,
        revision_number INTEGER NOT NULL DEFAULT 0,
        status TEXT NOT NULL DEFAULT 'open',
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(series_id, thread_key),
        CHECK(priority BETWEEN 0 AND 100),
        CHECK(status IN ('open', 'dormant', 'resolved', 'cancelled')),
        FOREIGN KEY (series_id) REFERENCES serialized_series_bibles(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_plot_threads_series ON serialized_plot_threads(series_id, status, priority DESC, last_advanced_episode_number DESC)`,
      `CREATE TABLE IF NOT EXISTS serialized_plot_thread_revisions (
        id TEXT PRIMARY KEY,
        plot_thread_id TEXT NOT NULL,
        series_id TEXT NOT NULL,
        episode_number INTEGER NOT NULL,
        revision_number INTEGER NOT NULL,
        snapshot TEXT NOT NULL DEFAULT '{}',
        change_kind TEXT NOT NULL DEFAULT 'episode_commit',
        change_reason TEXT NOT NULL DEFAULT '',
        actor TEXT,
        created_at TEXT NOT NULL,
        UNIQUE(plot_thread_id, revision_number),
        FOREIGN KEY (plot_thread_id) REFERENCES serialized_plot_threads(id) ON DELETE CASCADE,
        FOREIGN KEY (series_id) REFERENCES serialized_series_bibles(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_plot_thread_revisions_thread ON serialized_plot_thread_revisions(plot_thread_id, revision_number DESC)`,
      `CREATE TABLE IF NOT EXISTS serialized_plot_thread_episode_commits (
        id TEXT PRIMARY KEY,
        series_id TEXT NOT NULL,
        plot_thread_id TEXT NOT NULL,
        episode_memory_id TEXT NOT NULL,
        episode_number INTEGER NOT NULL,
        thread_revision_number INTEGER NOT NULL,
        commit_kind TEXT NOT NULL DEFAULT 'episode_commit',
        timeline_event_ids TEXT NOT NULL DEFAULT '[]',
        before_snapshot TEXT NOT NULL DEFAULT '{}',
        after_snapshot TEXT NOT NULL DEFAULT '{}',
        change_summary TEXT NOT NULL DEFAULT '',
        reason TEXT NOT NULL DEFAULT '',
        actor TEXT,
        created_at TEXT NOT NULL,
        FOREIGN KEY (series_id) REFERENCES serialized_series_bibles(id) ON DELETE CASCADE,
        FOREIGN KEY (plot_thread_id) REFERENCES serialized_plot_threads(id) ON DELETE CASCADE,
        FOREIGN KEY (episode_memory_id) REFERENCES serialized_episode_memories(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_plot_thread_commits_thread ON serialized_plot_thread_episode_commits(plot_thread_id, episode_number DESC, thread_revision_number DESC)`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_plot_thread_commits_series ON serialized_plot_thread_episode_commits(series_id, episode_number DESC, created_at DESC)`,
