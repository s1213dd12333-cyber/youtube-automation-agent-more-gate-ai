      `CREATE TABLE IF NOT EXISTS persistent_character_appearance_states (
        id TEXT PRIMARY KEY,
        character_id TEXT NOT NULL,
        production_id TEXT NOT NULL,
        scene_id TEXT,
        scope_key TEXT NOT NULL,
        version TEXT NOT NULL DEFAULT '11.11.4',
        status TEXT NOT NULL DEFAULT 'neutral',
        persistence TEXT NOT NULL DEFAULT 'scene',
        state_fingerprint TEXT NOT NULL,
        wardrobe TEXT,
        footwear TEXT,
        hair_state TEXT,
        condition TEXT,
        cleanliness TEXT,
        age_appearance TEXT,
        injuries TEXT NOT NULL DEFAULT '[]',
        carried_items TEXT NOT NULL DEFAULT '[]',
        temporary_accessories TEXT NOT NULL DEFAULT '[]',
        appearance_notes TEXT NOT NULL DEFAULT '[]',
        story_state TEXT NOT NULL DEFAULT '[]',

        source_ref TEXT,
        parent_state_id TEXT,

        canonical_identity_fingerprint TEXT,
