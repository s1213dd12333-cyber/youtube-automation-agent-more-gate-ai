  parseSerializedPlotThread(row) {
    if (!row) return null;
    const parse = (value, fallback = []) => { try { return JSON.parse(value || JSON.stringify(fallback)); } catch (_error) { return fallback; } };
    return {
      id: row.id,
      seriesId: row.series_id,
      threadKey: row.thread_key,
      title: row.title || '',
      threadType: row.thread_type || 'other',
      priority: Number(row.priority || 0),
      premise: row.premise || '',
      centralQuestion: row.central_question || '',
      stakes: row.stakes || '',
      currentState: row.current_state || '',
      openQuestions: parse(row.open_questions),
      resolvedQuestions: parse(row.resolved_questions),
      narrativePromises: parse(row.narrative_promises),
      establishedClues: parse(row.established_clues),
      redHerrings: parse(row.red_herrings),
      requiredPayoffs: parse(row.required_payoffs),
      involvedCharacterKeys: parse(row.involved_character_keys),
      relationshipEdgeKeys: parse(row.relationship_edge_keys),
      resolutionSummary: row.resolution_summary || '',
      notes: row.notes || '',
      introducedEpisodeNumber: Number(row.introduced_episode_number || 0),
      lastAdvancedEpisodeNumber: Number(row.last_advanced_episode_number || 0),
      resolvedEpisodeNumber: Number(row.resolved_episode_number || 0),
      revisionNumber: Number(row.revision_number || 0),
      status: row.status || 'open',
      createdAt: row.created_at,
      updatedAt: row.updated_at
    };
  }

  serializedPlotThreadSnapshot(thread = {}) {
    const stable = value => {
      if (Array.isArray(value)) return value.map(stable);
      if (!value || typeof value !== 'object') return value;
      return Object.keys(value).sort().reduce((out, key) => { out[key] = stable(value[key]); return out; }, {});
    };
    return stable({
      id: thread.id,
      seriesId: thread.seriesId,
      threadKey: thread.threadKey,
      title: thread.title || '',
      threadType: thread.threadType || 'other',
      priority: Number(thread.priority || 0),
      premise: thread.premise || '',
      centralQuestion: thread.centralQuestion || '',
      stakes: thread.stakes || '',
      currentState: thread.currentState || '',
      openQuestions: thread.openQuestions || [],
      resolvedQuestions: thread.resolvedQuestions || [],
      narrativePromises: thread.narrativePromises || [],
      establishedClues: thread.establishedClues || [],
      redHerrings: thread.redHerrings || [],
      requiredPayoffs: thread.requiredPayoffs || [],
      involvedCharacterKeys: thread.involvedCharacterKeys || [],
      relationshipEdgeKeys: thread.relationshipEdgeKeys || [],
      resolutionSummary: thread.resolutionSummary || '',
      notes: thread.notes || '',
      introducedEpisodeNumber: Number(thread.introducedEpisodeNumber || 0),
      lastAdvancedEpisodeNumber: Number(thread.lastAdvancedEpisodeNumber || 0),
      resolvedEpisodeNumber: Number(thread.resolvedEpisodeNumber || 0),
      revisionNumber: Number(thread.revisionNumber || 0),
      status: thread.status || 'open'
    });
  }

  async commitSerializedPlotThreadsAtomic(input = {}) {
    if (!input.seriesId || !input.episodeNumber || !input.episodeMemoryId) throw new Error('plot_thread_commit_identity_required');
    if (!Array.isArray(input.updates) || !input.updates.length) throw new Error('plot_thread_updates_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const series = await this.getRow('SELECT * FROM serialized_series_bibles WHERE id = ? LIMIT 1', [input.seriesId]);
      if (!series) throw new Error('plot_thread_series_not_found');
      if ((series.status || 'active') !== 'active') throw new Error('plot_thread_series_inactive');
      const episodeNumber = Number(input.episodeNumber);
      const currentEpisode = Number(series.current_episode || 0);
      if (episodeNumber > currentEpisode) throw new Error('plot_thread_future_episode_not_finalized');
      if (episodeNumber < currentEpisode && input.allowBackfill !== true) throw new Error('plot_thread_backfill_requires_explicit_approval');
      const memoryRow = await this.getRow(
        `SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? AND episode_number = ? AND status = 'finalized' LIMIT 1`,
        [input.episodeMemoryId, input.seriesId, episodeNumber]
      );
      if (!memoryRow) throw new Error('finalized_episode_memory_required');
      let memoryTimeline = []; try { memoryTimeline = JSON.parse(memoryRow.timeline_event_ids || '[]'); } catch (_error) {}
      const allowedTimeline = new Set(memoryTimeline);
      const now = new Date().toISOString();
      const threadIds = [];
      const commitIds = [];

      for (const item of input.updates) {
        const thread = item.thread || {};
        if (!thread.threadKey || !thread.title) throw new Error('plot_thread_identity_required');
        const row = await this.getRow(
          'SELECT * FROM serialized_plot_threads WHERE series_id = ? AND thread_key = ? LIMIT 1',
          [input.seriesId, thread.threadKey]
        );
        if (row && (row.status === 'resolved' || row.status === 'cancelled')) throw new Error('plot_thread_terminal_state_is_immutable');
        const currentRevision = Number(row?.revision_number || 0);
        if (Number(item.expectedRevision || 0) !== currentRevision) throw new Error('plot_thread_revision_conflict');
        const plotThreadId = row?.id || thread.id || this.generateId('serialized_plot_thread');
        const latest = row ? await this.getRow(
          `SELECT * FROM serialized_plot_thread_episode_commits WHERE plot_thread_id = ? ORDER BY thread_revision_number DESC LIMIT 1`,
          [plotThreadId]
        ) : null;
        if (latest && Number(latest.episode_number || 0) > episodeNumber) throw new Error('plot_thread_historical_insert_has_later_commit');
        const duplicate = row ? await this.getRow(
          `SELECT id FROM serialized_plot_thread_episode_commits
           WHERE plot_thread_id = ? AND episode_number = ? AND commit_kind = 'episode_commit' LIMIT 1`,
          [plotThreadId, episodeNumber]
        ) : null;
        if (duplicate) throw new Error('plot_thread_episode_already_committed');
        const timelineIds = [...new Set(item.relevantTimelineEventIds || [])];
        if (timelineIds.some(id => !allowedTimeline.has(id))) throw new Error('plot_thread_timeline_reference_outside_episode_memory');

        for (const characterKey of thread.involvedCharacterKeys || []) {
          const arc = await this.getRow(
            'SELECT * FROM serialized_character_arcs WHERE series_id = ? AND character_key = ? LIMIT 1',
            [input.seriesId, characterKey]
          );
          if (!arc) throw new Error('plot_thread_character_arc_not_found');
          if (Number(arc.first_episode_number || 0) > episodeNumber) throw new Error('plot_thread_character_not_introduced_yet');
        }
        for (const edgeKey of thread.relationshipEdgeKeys || []) {
          const parts = String(edgeKey || '').split('->');
          if (parts.length !== 2 || !parts[0] || !parts[1]) throw new Error('plot_thread_relationship_edge_invalid');
          const relationship = await this.getRow(
            `SELECT * FROM serialized_relationship_states
             WHERE series_id = ? AND source_character_key = ? AND target_character_key = ? LIMIT 1`,
            [input.seriesId, parts[0], parts[1]]
          );
          if (!relationship) throw new Error('plot_thread_relationship_edge_not_found');
          if (Number(relationship.first_episode_number || 0) > episodeNumber) throw new Error('plot_thread_relationship_not_established_yet');
        }
        const terminal = thread.status === 'resolved' || thread.status === 'cancelled';
        if (terminal && String(thread.resolutionSummary || '').trim().length < 8) throw new Error('plot_thread_resolution_summary_required');

        const before = row ? this.serializedPlotThreadSnapshot(this.parseSerializedPlotThread(row)) : {};
        const nextRevision = currentRevision + 1;
        const introducedEpisode = row ? Number(row.introduced_episode_number || episodeNumber) : episodeNumber;
        const resolvedEpisode = terminal ? episodeNumber : 0;
        const values = [
          thread.title || '', thread.threadType || 'other', Number(thread.priority || 0), thread.premise || '',
          thread.centralQuestion || '', thread.stakes || '', thread.currentState || '', JSON.stringify(thread.openQuestions || []),
          JSON.stringify(thread.resolvedQuestions || []), JSON.stringify(thread.narrativePromises || []),
          JSON.stringify(thread.establishedClues || []), JSON.stringify(thread.redHerrings || []), JSON.stringify(thread.requiredPayoffs || []),
          JSON.stringify(thread.involvedCharacterKeys || []), JSON.stringify(thread.relationshipEdgeKeys || []),
          thread.resolutionSummary || '', thread.notes || ''
        ];
        if (!row) {
          await this.executeQuery(
            `INSERT INTO serialized_plot_threads (
              id, series_id, thread_key, title, thread_type, priority, premise, central_question, stakes, current_state,
              open_questions, resolved_questions, narrative_promises, established_clues, red_herrings, required_payoffs,
              involved_character_keys, relationship_edge_keys, resolution_summary, notes, introduced_episode_number,
              last_advanced_episode_number, resolved_episode_number, revision_number, status, created_at, updated_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
            [plotThreadId, input.seriesId, thread.threadKey, ...values, introducedEpisode, episodeNumber, resolvedEpisode,
              nextRevision, thread.status || 'open', now, now]
          );
        } else {
          await this.executeQuery(
            `UPDATE serialized_plot_threads SET
              title = ?, thread_type = ?, priority = ?, premise = ?, central_question = ?, stakes = ?, current_state = ?,
              open_questions = ?, resolved_questions = ?, narrative_promises = ?, established_clues = ?, red_herrings = ?,
              required_payoffs = ?, involved_character_keys = ?, relationship_edge_keys = ?, resolution_summary = ?, notes = ?,
              last_advanced_episode_number = ?, resolved_episode_number = ?, revision_number = ?, status = ?, updated_at = ?
             WHERE id = ? AND series_id = ? AND revision_number = ?`,
            [...values, episodeNumber, resolvedEpisode, nextRevision, thread.status || 'open', now,
              plotThreadId, input.seriesId, currentRevision]
          );
          const check = await this.getRow('SELECT revision_number FROM serialized_plot_threads WHERE id = ? LIMIT 1', [plotThreadId]);
          if (Number(check?.revision_number || -1) !== nextRevision) throw new Error('plot_thread_revision_conflict');
        }
        const saved = await this.getSerializedPlotThreadById(plotThreadId);
        const after = this.serializedPlotThreadSnapshot(saved);
        await this.executeQuery(
          `INSERT INTO serialized_plot_thread_revisions (
            id, plot_thread_id, series_id, episode_number, revision_number, snapshot, change_kind, change_reason, actor, created_at
          ) VALUES (?, ?, ?, ?, ?, ?, 'episode_commit', ?, ?, ?)`,
          [this.generateId('serialized_plot_thread_revision'), plotThreadId, input.seriesId, episodeNumber, nextRevision,
            JSON.stringify(after), input.reason || '', input.actor || null, now]
        );
        const commitId = this.generateId('serialized_plot_thread_commit');
        await this.executeQuery(
          `INSERT INTO serialized_plot_thread_episode_commits (
            id, series_id, plot_thread_id, episode_memory_id, episode_number, thread_revision_number, commit_kind,
            timeline_event_ids, before_snapshot, after_snapshot, change_summary, reason, actor, created_at
          ) VALUES (?, ?, ?, ?, ?, ?, 'episode_commit', ?, ?, ?, ?, ?, ?, ?)`,
          [commitId, input.seriesId, plotThreadId, input.episodeMemoryId, episodeNumber, nextRevision,
            JSON.stringify(timelineIds), JSON.stringify(before), JSON.stringify(after), item.changeSummary || '',
            input.reason || '', input.actor || null, now]
        );
        threadIds.push(plotThreadId);
        commitIds.push(commitId);
      }
      await this.executeQuery('COMMIT');
      const threads = [];
      for (const id of threadIds) threads.push(await this.getSerializedPlotThreadById(id));
      const commits = [];
      for (const id of commitIds) commits.push(this.parseSerializedPlotThreadCommit(await this.getRow(
        'SELECT * FROM serialized_plot_thread_episode_commits WHERE id = ? LIMIT 1', [id]
      )));
      return { threads, commits };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }

  async amendSerializedPlotThreadAtomic(input = {}) {
    if (!input.seriesId || !input.thread?.id || !input.episodeMemoryId) throw new Error('plot_thread_amendment_identity_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const row = await this.getRow(
        'SELECT * FROM serialized_plot_threads WHERE id = ? AND series_id = ? LIMIT 1', [input.thread.id, input.seriesId]
      );
      if (!row) throw new Error('plot_thread_not_found');
      const currentRevision = Number(row.revision_number || 0);
      if (Number(input.expectedRevision) !== currentRevision) throw new Error('plot_thread_revision_conflict');
      if (row.thread_key !== input.thread.threadKey) throw new Error('plot_thread_key_is_immutable');
      const latest = await this.getRow(
        `SELECT * FROM serialized_plot_thread_episode_commits WHERE plot_thread_id = ? ORDER BY thread_revision_number DESC LIMIT 1`,
        [input.thread.id]
      );
      if (!latest || Number(latest.episode_number || 0) !== Number(input.episodeNumber)) {
        throw new Error('historical_plot_thread_amendment_blocked_by_later_state');
      }
      const memory = await this.getRow(
        `SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? AND episode_number = ? AND status = 'finalized' LIMIT 1`,
        [input.episodeMemoryId, input.seriesId, Number(input.episodeNumber)]
      );
      if (!memory) throw new Error('finalized_episode_memory_required');
      let allowedIds = []; try { allowedIds = JSON.parse(memory.timeline_event_ids || '[]'); } catch (_error) {}
      const allowed = new Set(allowedIds);
      const timelineIds = [...new Set(input.relevantTimelineEventIds || [])];
      if (timelineIds.some(id => !allowed.has(id))) throw new Error('plot_thread_timeline_reference_outside_episode_memory');

      const thread = input.thread;
      for (const characterKey of thread.involvedCharacterKeys || []) {
        const arc = await this.getRow('SELECT * FROM serialized_character_arcs WHERE series_id = ? AND character_key = ? LIMIT 1', [input.seriesId, characterKey]);
        if (!arc) throw new Error('plot_thread_character_arc_not_found');
        if (Number(arc.first_episode_number || 0) > Number(input.episodeNumber)) throw new Error('plot_thread_character_not_introduced_yet');
      }
      for (const edgeKey of thread.relationshipEdgeKeys || []) {
        const parts = String(edgeKey || '').split('->');
        const relationship = parts.length === 2 ? await this.getRow(
          `SELECT * FROM serialized_relationship_states WHERE series_id = ? AND source_character_key = ? AND target_character_key = ? LIMIT 1`,
          [input.seriesId, parts[0], parts[1]]
        ) : null;
        if (!relationship) throw new Error('plot_thread_relationship_edge_not_found');
        if (Number(relationship.first_episode_number || 0) > Number(input.episodeNumber)) throw new Error('plot_thread_relationship_not_established_yet');
      }
      const terminal = thread.status === 'resolved' || thread.status === 'cancelled';
      if (terminal && String(thread.resolutionSummary || '').trim().length < 8) throw new Error('plot_thread_resolution_summary_required');

      const before = this.serializedPlotThreadSnapshot(this.parseSerializedPlotThread(row));
      const nextRevision = currentRevision + 1;
      const now = new Date().toISOString();
      const resolvedEpisode = terminal ? Number(input.episodeNumber) : 0;
      await this.executeQuery(
        `UPDATE serialized_plot_threads SET
          title = ?, thread_type = ?, priority = ?, premise = ?, central_question = ?, stakes = ?, current_state = ?,
          open_questions = ?, resolved_questions = ?, narrative_promises = ?, established_clues = ?, red_herrings = ?,
          required_payoffs = ?, involved_character_keys = ?, relationship_edge_keys = ?, resolution_summary = ?, notes = ?,
          resolved_episode_number = ?, revision_number = ?, status = ?, updated_at = ?
         WHERE id = ? AND series_id = ? AND revision_number = ?`,
        [thread.title || '', thread.threadType || 'other', Number(thread.priority || 0), thread.premise || '',
          thread.centralQuestion || '', thread.stakes || '', thread.currentState || '', JSON.stringify(thread.openQuestions || []),
          JSON.stringify(thread.resolvedQuestions || []), JSON.stringify(thread.narrativePromises || []),
          JSON.stringify(thread.establishedClues || []), JSON.stringify(thread.redHerrings || []), JSON.stringify(thread.requiredPayoffs || []),
          JSON.stringify(thread.involvedCharacterKeys || []), JSON.stringify(thread.relationshipEdgeKeys || []), thread.resolutionSummary || '',
          thread.notes || '', resolvedEpisode, nextRevision, thread.status || 'open', now, thread.id, input.seriesId, currentRevision]
      );
      const saved = await this.getSerializedPlotThreadById(thread.id);
      if (Number(saved?.revisionNumber || -1) !== nextRevision) throw new Error('plot_thread_revision_conflict');
      const after = this.serializedPlotThreadSnapshot(saved);
      await this.executeQuery(
        `INSERT INTO serialized_plot_thread_revisions (
          id, plot_thread_id, series_id, episode_number, revision_number, snapshot, change_kind, change_reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?)`,
        [this.generateId('serialized_plot_thread_revision'), thread.id, input.seriesId, Number(input.episodeNumber), nextRevision,
          JSON.stringify(after), input.reason || '', input.actor || null, now]
      );
      const commitId = this.generateId('serialized_plot_thread_commit');
      await this.executeQuery(
        `INSERT INTO serialized_plot_thread_episode_commits (
          id, series_id, plot_thread_id, episode_memory_id, episode_number, thread_revision_number, commit_kind,
          timeline_event_ids, before_snapshot, after_snapshot, change_summary, reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?, ?, ?, ?, ?)`,
        [commitId, input.seriesId, thread.id, input.episodeMemoryId, Number(input.episodeNumber), nextRevision,
          JSON.stringify(timelineIds), JSON.stringify(before), JSON.stringify(after), input.changeSummary || '', input.reason || '',
          input.actor || null, now]
      );
      await this.executeQuery('COMMIT');
      return {
        thread: await this.getSerializedPlotThreadById(thread.id),
        commit: this.parseSerializedPlotThreadCommit(await this.getRow(
          'SELECT * FROM serialized_plot_thread_episode_commits WHERE id = ? LIMIT 1', [commitId]
        ))
      };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }
