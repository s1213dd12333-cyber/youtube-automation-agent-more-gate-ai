  parseSerializedCharacterArc(row) {
    if (!row) return null;
    const parse = (value, fallback = []) => { try { return JSON.parse(value || JSON.stringify(fallback)); } catch (_error) { return fallback; } };
    return {
      id: row.id,
      seriesId: row.series_id,
      characterKey: row.character_key,
      displayName: row.display_name,
      persistentCharacterId: row.persistent_character_id || null,
      persistentCharacterFingerprint: row.persistent_character_fingerprint || null,
      arcPhase: row.arc_phase || '',
      storyStatus: row.story_status || 'active',
      emotionalState: row.emotional_state || '',
      moralState: row.moral_state || '',
      physicalCondition: row.physical_condition || '',
      goals: parse(row.goals), motivations: parse(row.motivations), beliefs: parse(row.beliefs),
      knowledge: parse(row.knowledge), secrets: parse(row.secrets), innerConflicts: parse(row.inner_conflicts),
      commitments: parse(row.commitments), milestones: parse(row.milestones), notes: row.notes || '',
      firstEpisodeNumber: Number(row.first_episode_number || 0),
      lastEpisodeNumber: Number(row.last_episode_number || 0),
      revisionNumber: Number(row.revision_number || 0),
      status: row.status || 'active', createdAt: row.created_at, updatedAt: row.updated_at
    };
  }

  serializedCharacterArcSnapshot(arc = {}) {
    const stable = value => {
      if (Array.isArray(value)) return value.map(stable);
      if (!value || typeof value !== 'object') return value;
      return Object.keys(value).sort().reduce((out, key) => { out[key] = stable(value[key]); return out; }, {});
    };
    return stable({
      id: arc.id, seriesId: arc.seriesId, characterKey: arc.characterKey, displayName: arc.displayName,
      persistentCharacterId: arc.persistentCharacterId || null,
      persistentCharacterFingerprint: arc.persistentCharacterFingerprint || null,
      arcPhase: arc.arcPhase || '', storyStatus: arc.storyStatus || 'active',
      emotionalState: arc.emotionalState || '', moralState: arc.moralState || '', physicalCondition: arc.physicalCondition || '',
      goals: arc.goals || [], motivations: arc.motivations || [], beliefs: arc.beliefs || [], knowledge: arc.knowledge || [],
      secrets: arc.secrets || [], innerConflicts: arc.innerConflicts || [], commitments: arc.commitments || [],
      milestones: arc.milestones || [], notes: arc.notes || '', firstEpisodeNumber: Number(arc.firstEpisodeNumber || 0),
      lastEpisodeNumber: Number(arc.lastEpisodeNumber || 0), revisionNumber: Number(arc.revisionNumber || 0), status: arc.status || 'active'
    });
  }

  async commitSerializedCharacterArcsAtomic(input = {}) {
    if (!input.seriesId || !input.episodeNumber || !input.episodeMemoryId) throw new Error('character_arc_commit_identity_required');
    if (!Array.isArray(input.updates) || !input.updates.length) throw new Error('character_arc_updates_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const series = await this.getRow('SELECT * FROM serialized_series_bibles WHERE id = ? LIMIT 1', [input.seriesId]);
      if (!series) throw new Error('character_arc_series_not_found');
      if ((series.status || 'active') !== 'active') throw new Error('character_arc_series_inactive');
      const episodeNumber = Number(input.episodeNumber);
      const currentEpisode = Number(series.current_episode || 0);
      if (episodeNumber > currentEpisode) throw new Error('character_arc_future_episode_not_finalized');
      if (episodeNumber < currentEpisode && input.allowBackfill !== true) throw new Error('character_arc_backfill_requires_explicit_approval');
      const memoryRow = await this.getRow(
        `SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? AND episode_number = ? AND status = 'finalized' LIMIT 1`,
        [input.episodeMemoryId, input.seriesId, episodeNumber]
      );
      if (!memoryRow) throw new Error('finalized_episode_memory_required');
      let memoryTimeline = []; try { memoryTimeline = JSON.parse(memoryRow.timeline_event_ids || '[]'); } catch (_error) {}
      const allowedTimeline = new Set(memoryTimeline);
      const now = new Date().toISOString();
      const commitIds = [];
      const arcIds = [];

      for (const item of input.updates) {
        const arc = item.arc || {};
        const row = await this.getRow(
          'SELECT * FROM serialized_character_arcs WHERE series_id = ? AND character_key = ? LIMIT 1',
          [input.seriesId, arc.characterKey]
        );
        const currentRevision = Number(row?.revision_number || 0);
        if (Number(item.expectedRevision || 0) !== currentRevision) throw new Error('character_arc_revision_conflict');
        const arcId = row?.id || arc.id || this.generateId('serialized_character_arc');
        const latest = row ? await this.getRow(
          `SELECT * FROM serialized_character_arc_episode_commits WHERE character_arc_id = ? ORDER BY arc_revision_number DESC LIMIT 1`, [arcId]
        ) : null;
        if (latest && Number(latest.episode_number || 0) > episodeNumber) throw new Error('character_arc_historical_insert_has_later_commit');
        const duplicate = row ? await this.getRow(
          `SELECT id FROM serialized_character_arc_episode_commits WHERE character_arc_id = ? AND episode_number = ? AND commit_kind = 'episode_commit' LIMIT 1`,
          [arcId, episodeNumber]
        ) : null;
        if (duplicate) throw new Error('character_arc_episode_already_committed');
        const timelineIds = [...new Set(item.relevantTimelineEventIds || [])];
        if (timelineIds.some(id => !allowedTimeline.has(id))) throw new Error('character_arc_timeline_reference_outside_episode_memory');

        if (arc.persistentCharacterId) {
          const visual = await this.getRow('SELECT * FROM persistent_characters WHERE id = ? LIMIT 1', [arc.persistentCharacterId]);
          if (!visual || (visual.status || 'active') !== 'active') throw new Error('character_arc_visual_binding_invalid');
          if (arc.persistentCharacterFingerprint && visual.identity_fingerprint !== arc.persistentCharacterFingerprint) throw new Error('character_arc_visual_fingerprint_drift');
          if (row?.persistent_character_id && row.persistent_character_id !== arc.persistentCharacterId && item.allowRebind !== true) {
            throw new Error('character_arc_visual_rebind_requires_explicit_approval');
          }
        }

        const before = row ? this.serializedCharacterArcSnapshot(this.parseSerializedCharacterArc(row)) : {};
        const nextRevision = currentRevision + 1;
        const firstEpisode = row ? Number(row.first_episode_number || episodeNumber) : episodeNumber;
        if (!row) {
          await this.executeQuery(
            `INSERT INTO serialized_character_arcs (
              id, series_id, character_key, display_name, persistent_character_id, persistent_character_fingerprint,
              arc_phase, story_status, emotional_state, moral_state, physical_condition, goals, motivations, beliefs,
              knowledge, secrets, inner_conflicts, commitments, milestones, notes, first_episode_number,
              last_episode_number, revision_number, status, created_at, updated_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
            [arcId, input.seriesId, arc.characterKey, arc.displayName, arc.persistentCharacterId || null,
              arc.persistentCharacterFingerprint || null, arc.arcPhase || '', arc.storyStatus || 'active',
              arc.emotionalState || '', arc.moralState || '', arc.physicalCondition || '', JSON.stringify(arc.goals || []),
              JSON.stringify(arc.motivations || []), JSON.stringify(arc.beliefs || []), JSON.stringify(arc.knowledge || []),
              JSON.stringify(arc.secrets || []), JSON.stringify(arc.innerConflicts || []), JSON.stringify(arc.commitments || []),
              JSON.stringify(arc.milestones || []), arc.notes || '', firstEpisode, episodeNumber, nextRevision,
              arc.status || 'active', now, now]
          );
        } else {
          await this.executeQuery(
            `UPDATE serialized_character_arcs SET display_name = ?, persistent_character_id = ?, persistent_character_fingerprint = ?,
              arc_phase = ?, story_status = ?, emotional_state = ?, moral_state = ?, physical_condition = ?, goals = ?, motivations = ?,
              beliefs = ?, knowledge = ?, secrets = ?, inner_conflicts = ?, commitments = ?, milestones = ?, notes = ?,
              last_episode_number = ?, revision_number = ?, status = ?, updated_at = ?
             WHERE id = ? AND series_id = ? AND revision_number = ?`,
            [arc.displayName, arc.persistentCharacterId || null, arc.persistentCharacterFingerprint || null, arc.arcPhase || '',
              arc.storyStatus || 'active', arc.emotionalState || '', arc.moralState || '', arc.physicalCondition || '',
              JSON.stringify(arc.goals || []), JSON.stringify(arc.motivations || []), JSON.stringify(arc.beliefs || []),
              JSON.stringify(arc.knowledge || []), JSON.stringify(arc.secrets || []), JSON.stringify(arc.innerConflicts || []),
              JSON.stringify(arc.commitments || []), JSON.stringify(arc.milestones || []), arc.notes || '', episodeNumber,
              nextRevision, arc.status || 'active', now, arcId, input.seriesId, currentRevision]
          );
          const check = await this.getRow('SELECT revision_number FROM serialized_character_arcs WHERE id = ? LIMIT 1', [arcId]);
          if (Number(check?.revision_number || -1) !== nextRevision) throw new Error('character_arc_revision_conflict');
        }
        const saved = await this.getSerializedCharacterArcById(arcId);
        const after = this.serializedCharacterArcSnapshot(saved);
        await this.executeQuery(
          `INSERT INTO serialized_character_arc_revisions (
            id, character_arc_id, series_id, episode_number, revision_number, snapshot, change_kind, change_reason, actor, created_at
          ) VALUES (?, ?, ?, ?, ?, ?, 'episode_commit', ?, ?, ?)`,
          [this.generateId('serialized_character_arc_revision'), arcId, input.seriesId, episodeNumber, nextRevision,
            JSON.stringify(after), input.reason || '', input.actor || null, now]
        );
        const commitId = this.generateId('serialized_character_arc_commit');
        await this.executeQuery(
          `INSERT INTO serialized_character_arc_episode_commits (
            id, series_id, character_arc_id, episode_memory_id, episode_number, arc_revision_number, commit_kind,
            timeline_event_ids, before_snapshot, after_snapshot, change_summary, reason, actor, created_at
          ) VALUES (?, ?, ?, ?, ?, ?, 'episode_commit', ?, ?, ?, ?, ?, ?, ?)`,
          [commitId, input.seriesId, arcId, input.episodeMemoryId, episodeNumber, nextRevision,
            JSON.stringify(timelineIds), JSON.stringify(before), JSON.stringify(after), item.changeSummary || '',
            input.reason || '', input.actor || null, now]
        );
        arcIds.push(arcId); commitIds.push(commitId);
      }
      await this.executeQuery('COMMIT');
      const arcs = []; for (const id of arcIds) arcs.push(await this.getSerializedCharacterArcById(id));
      const commits = []; for (const id of commitIds) commits.push(this.parseSerializedCharacterArcCommit(await this.getRow(
        'SELECT * FROM serialized_character_arc_episode_commits WHERE id = ? LIMIT 1', [id]
      )));
      return { arcs, commits };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }

  async amendSerializedCharacterArcAtomic(input = {}) {
    if (!input.seriesId || !input.arc?.id || !input.episodeMemoryId) throw new Error('character_arc_amendment_identity_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const row = await this.getRow('SELECT * FROM serialized_character_arcs WHERE id = ? AND series_id = ? LIMIT 1', [input.arc.id, input.seriesId]);
      if (!row) throw new Error('character_arc_not_found');
      const currentRevision = Number(row.revision_number || 0);
      if (Number(input.expectedRevision) !== currentRevision) throw new Error('character_arc_revision_conflict');
      const latest = await this.getRow(
        `SELECT * FROM serialized_character_arc_episode_commits WHERE character_arc_id = ? ORDER BY arc_revision_number DESC LIMIT 1`, [input.arc.id]
      );
      if (!latest || Number(latest.episode_number || 0) !== Number(input.episodeNumber)) throw new Error('historical_character_arc_amendment_blocked_by_later_state');
      const memory = await this.getRow(
        `SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? AND episode_number = ? AND status = 'finalized' LIMIT 1`,
        [input.episodeMemoryId, input.seriesId, Number(input.episodeNumber)]
      );
      if (!memory) throw new Error('finalized_episode_memory_required');
      let allowedIds = []; try { allowedIds = JSON.parse(memory.timeline_event_ids || '[]'); } catch (_error) {}
      const allowed = new Set(allowedIds);
      const timelineIds = [...new Set(input.relevantTimelineEventIds || [])];
      if (timelineIds.some(id => !allowed.has(id))) throw new Error('character_arc_timeline_reference_outside_episode_memory');
      if (input.arc.persistentCharacterId) {
        const visual = await this.getRow('SELECT * FROM persistent_characters WHERE id = ? LIMIT 1', [input.arc.persistentCharacterId]);
        if (!visual || (visual.status || 'active') !== 'active') throw new Error('character_arc_visual_binding_invalid');
        if (input.arc.persistentCharacterFingerprint && visual.identity_fingerprint !== input.arc.persistentCharacterFingerprint) throw new Error('character_arc_visual_fingerprint_drift');
      }
      const before = this.serializedCharacterArcSnapshot(this.parseSerializedCharacterArc(row));
      const nextRevision = currentRevision + 1;
      const now = new Date().toISOString();
      await this.executeQuery(
        `UPDATE serialized_character_arcs SET display_name = ?, persistent_character_id = ?, persistent_character_fingerprint = ?,
          arc_phase = ?, story_status = ?, emotional_state = ?, moral_state = ?, physical_condition = ?, goals = ?, motivations = ?,
          beliefs = ?, knowledge = ?, secrets = ?, inner_conflicts = ?, commitments = ?, milestones = ?, notes = ?,
          revision_number = ?, status = ?, updated_at = ? WHERE id = ? AND series_id = ? AND revision_number = ?`,
        [input.arc.displayName, input.arc.persistentCharacterId || null, input.arc.persistentCharacterFingerprint || null,
          input.arc.arcPhase || '', input.arc.storyStatus || 'active', input.arc.emotionalState || '', input.arc.moralState || '',
          input.arc.physicalCondition || '', JSON.stringify(input.arc.goals || []), JSON.stringify(input.arc.motivations || []),
          JSON.stringify(input.arc.beliefs || []), JSON.stringify(input.arc.knowledge || []), JSON.stringify(input.arc.secrets || []),
          JSON.stringify(input.arc.innerConflicts || []), JSON.stringify(input.arc.commitments || []), JSON.stringify(input.arc.milestones || []),
          input.arc.notes || '', nextRevision, input.arc.status || 'active', now, input.arc.id, input.seriesId, currentRevision]
      );
      const saved = await this.getSerializedCharacterArcById(input.arc.id);
      if (Number(saved?.revisionNumber || -1) !== nextRevision) throw new Error('character_arc_revision_conflict');
      const after = this.serializedCharacterArcSnapshot(saved);
      await this.executeQuery(
        `INSERT INTO serialized_character_arc_revisions (
          id, character_arc_id, series_id, episode_number, revision_number, snapshot, change_kind, change_reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?)`,

          nextRevision, JSON.stringify(after), input.reason || '', input.actor || null, now]
      );
      const commitId = this.generateId('serialized_character_arc_commit');
      await this.executeQuery(

          id, series_id, character_arc_id, episode_memory_id, episode_number, arc_revision_number, commit_kind,
          timeline_event_ids, before_snapshot, after_snapshot, change_summary, reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?, ?, ?, ?, ?)`,
        [commitId, input.seriesId, input.arc.id, input.episodeMemoryId, Number(input.episodeNumber), nextRevision,
          JSON.stringify(timelineIds), JSON.stringify(before), JSON.stringify(after), input.changeSummary || '', input.reason || '',
          input.actor || null, now]
      );
      await this.executeQuery('COMMIT');
      return { arc: await this.getSerializedCharacterArcById(input.arc.id), commit: this.parseSerializedCharacterArcCommit(await this.getRow(
        'SELECT * FROM serialized_character_arc_episode_commits WHERE id = ? LIMIT 1', [commitId]
      )) };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
