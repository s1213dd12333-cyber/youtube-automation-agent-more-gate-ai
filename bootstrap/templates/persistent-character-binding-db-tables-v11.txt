      // Phase 11.11.5 explicit scene/shot bindings for persistent characters
      `CREATE TABLE IF NOT EXISTS persistent_character_bindings (
        id TEXT PRIMARY KEY,
        production_id TEXT NOT NULL,
        scene_id TEXT NOT NULL,
        shot_id TEXT NOT NULL,
        shot_index INTEGER NOT NULL DEFAULT 0,
        character_id TEXT,
        character_key TEXT,
        version TEXT NOT NULL DEFAULT '11.11.5',
        status TEXT NOT NULL,
        reference_text TEXT,
        reference_key TEXT,
        match_mode TEXT,
        confidence REAL NOT NULL DEFAULT 0,
        visibility TEXT NOT NULL DEFAULT 'visible',
        placement TEXT,
        action TEXT,
        expression TEXT,
        required INTEGER NOT NULL DEFAULT 1,
        state_id TEXT,
        state_fingerprint TEXT,
        canonical_asset_id TEXT,
        canonical_asset_sha256 TEXT,
        binding_fingerprint TEXT NOT NULL,
        prompt_fragment TEXT NOT NULL DEFAULT '',
        source_kind TEXT NOT NULL DEFAULT 'scene_character_binding',
        source_ref TEXT,
        reason TEXT,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(production_id, scene_id, shot_id, id),
        FOREIGN KEY (production_id) REFERENCES productions(id) ON DELETE CASCADE,
        FOREIGN KEY (scene_id) REFERENCES production_scenes(id) ON DELETE CASCADE,
        FOREIGN KEY (character_id) REFERENCES persistent_characters(id) ON DELETE SET NULL,
        FOREIGN KEY (state_id) REFERENCES persistent_character_appearance_states(id) ON DELETE SET NULL,
        FOREIGN KEY (canonical_asset_id) REFERENCES persistent_character_assets(id) ON DELETE SET NULL
      )`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_character_bindings_prod ON persistent_character_bindings(production_id, scene_id, shot_index)`,
      `CREATE INDEX IF NOT EXISTS idx_persistent_character_bindings_character ON persistent_character_bindings(character_id, production_id, scene_id)`,
