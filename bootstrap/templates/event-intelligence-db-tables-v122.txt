      // Phase 12.2 — Event Intelligence Engine: stable event identity, revisions and non-causal relations.
      `CREATE TABLE IF NOT EXISTS global_news_events (
        id TEXT PRIMARY KEY,
        canonical_title TEXT NOT NULL,
        status TEXT NOT NULL DEFAULT 'active' CHECK(status IN ('active','dormant','resolved','archived')),
        first_seen_at TEXT NOT NULL,
        last_seen_at TEXT NOT NULL,
        last_material_change TEXT,
        revision_number INTEGER NOT NULL DEFAULT 0,
        signature_json TEXT NOT NULL DEFAULT '{}',
        entities_json TEXT NOT NULL DEFAULT '[]',
        locations_json TEXT NOT NULL DEFAULT '[]',
        concepts_json TEXT NOT NULL DEFAULT '[]',
        numbers_json TEXT NOT NULL DEFAULT '[]',
        languages_json TEXT NOT NULL DEFAULT '[]',
        title_variants_json TEXT NOT NULL DEFAULT '[]',
        descriptor_fingerprint TEXT NOT NULL,
        primary_cluster_id TEXT NOT NULL,
        latest_cluster_id TEXT NOT NULL,
        source_count INTEGER NOT NULL DEFAULT 0,
        article_count INTEGER NOT NULL DEFAULT 0,
        observation_count INTEGER NOT NULL DEFAULT 0,
        confidence_score INTEGER NOT NULL DEFAULT 0 CHECK(confidence_score BETWEEN 0 AND 100),
        evolution_score INTEGER NOT NULL DEFAULT 0 CHECK(evolution_score BETWEEN 0 AND 100),
        created_at TEXT DEFAULT CURRENT_TIMESTAMP,
        updated_at TEXT DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (primary_cluster_id) REFERENCES global_news_clusters(id),
        FOREIGN KEY (latest_cluster_id) REFERENCES global_news_clusters(id)
      )`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_events_recent ON global_news_events(last_material_change DESC, last_seen_at DESC)`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_events_status ON global_news_events(status, last_seen_at DESC)`,
      `CREATE TABLE IF NOT EXISTS global_news_event_revisions (
        id TEXT PRIMARY KEY,
        event_id TEXT NOT NULL,
        revision_number INTEGER NOT NULL,
        change_kind TEXT NOT NULL CHECK(change_kind IN ('new_event','observation','material_update','escalation','resolution','correction')),
        cluster_id TEXT NOT NULL,
        scan_id TEXT,
        before_json TEXT NOT NULL DEFAULT '{}',
        after_json TEXT NOT NULL DEFAULT '{}',
        delta_json TEXT NOT NULL DEFAULT '{}',
        descriptor_fingerprint TEXT NOT NULL,
        created_at TEXT DEFAULT CURRENT_TIMESTAMP,
        UNIQUE(event_id, revision_number),
        FOREIGN KEY (event_id) REFERENCES global_news_events(id),
        FOREIGN KEY (cluster_id) REFERENCES global_news_clusters(id),
        FOREIGN KEY (scan_id) REFERENCES global_news_scans(id)
      )`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_event_revisions_event ON global_news_event_revisions(event_id, revision_number DESC)`,
      `CREATE TABLE IF NOT EXISTS global_news_event_cluster_links (
        event_id TEXT NOT NULL,
        cluster_id TEXT NOT NULL UNIQUE,
        match_score REAL NOT NULL DEFAULT 0 CHECK(match_score BETWEEN 0 AND 1),
        match_reasons_json TEXT NOT NULL DEFAULT '{}',
        first_scan_id TEXT,
        last_scan_id TEXT,
        first_linked_at TEXT DEFAULT CURRENT_TIMESTAMP,
        last_seen_at TEXT DEFAULT CURRENT_TIMESTAMP,
        PRIMARY KEY(event_id, cluster_id),
        FOREIGN KEY (event_id) REFERENCES global_news_events(id),
        FOREIGN KEY (cluster_id) REFERENCES global_news_clusters(id),
        FOREIGN KEY (first_scan_id) REFERENCES global_news_scans(id),
        FOREIGN KEY (last_scan_id) REFERENCES global_news_scans(id)
      )`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_event_links_event ON global_news_event_cluster_links(event_id, last_seen_at DESC)`,
      `CREATE TABLE IF NOT EXISTS global_news_event_relations (
        id TEXT PRIMARY KEY,
        source_event_id TEXT NOT NULL,
        target_event_id TEXT NOT NULL,
        relation_type TEXT NOT NULL CHECK(relation_type IN ('related_to','follow_up_of')),
        confidence INTEGER NOT NULL DEFAULT 0 CHECK(confidence BETWEEN 0 AND 100),
        evidence_json TEXT NOT NULL DEFAULT '{}',
        created_at TEXT DEFAULT CURRENT_TIMESTAMP,
        UNIQUE(source_event_id, target_event_id, relation_type),
        CHECK(source_event_id != target_event_id),
        FOREIGN KEY (source_event_id) REFERENCES global_news_events(id),
        FOREIGN KEY (target_event_id) REFERENCES global_news_events(id)
      )`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_event_relations_source ON global_news_event_relations(source_event_id, confidence DESC)`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_event_relations_target ON global_news_event_relations(target_event_id, confidence DESC)`,
      `CREATE TABLE IF NOT EXISTS global_news_event_assignments (
        id TEXT PRIMARY KEY,
        event_id TEXT NOT NULL,
        decision_id TEXT NOT NULL UNIQUE,
        assignment_id TEXT,
        idea_id TEXT,
        created_at TEXT DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (event_id) REFERENCES global_news_events(id),
        FOREIGN KEY (decision_id) REFERENCES global_news_editor_decisions(id),

        FOREIGN KEY (idea_id) REFERENCES content_ideas(id)
      )`,
      `CREATE INDEX IF NOT EXISTS idx_global_news_event_assignments_event ON global_news_event_assignments(event_id, created_at DESC)`,
