  parseSerializedEpisodeMemory(row) {
    if (!row) return null;
    const parse = (value, fallback = []) => {
      try { return JSON.parse(value || JSON.stringify(fallback)); } catch (_error) { return fallback; }
    };
    return {
      id: row.id,
      seriesId: row.series_id,
      episodeNumber: Number(row.episode_number || 0),
      strategyId: row.strategy_id || null,
      productionId: row.production_id || null,
      title: row.title || '',
      summary: row.summary || '',
      discoveries: parse(row.discoveries, []),
      establishedFacts: parse(row.established_facts, []),
      resolvedQuestions: parse(row.resolved_questions, []),
      unresolvedQuestions: parse(row.unresolved_questions, []),
      narrativePromises: parse(row.narrative_promises, []),
      cliffhangers: parse(row.cliffhangers, []),
      timelineEventIds: parse(row.timeline_event_ids, []),
      approvalId: row.approval_id,
      approvedBy: row.approved_by,
      approvalReason: row.approval_reason,
      approvedAt: row.approved_at,
      sourceFingerprint: row.source_fingerprint || null,
      revisionNumber: Number(row.revision_number || 1),
      status: row.status || 'finalized',
      createdAt: row.created_at,
      updatedAt: row.updated_at
    };
  }

  async getSerializedEpisodeMemoryById(memoryId) {
    const row = await this.getRow('SELECT * FROM serialized_episode_memories WHERE id = ? LIMIT 1', [memoryId]);
    return this.parseSerializedEpisodeMemory(row);
  }

  async getSerializedEpisodeMemory(seriesId, episodeNumber) {
    const row = await this.getRow(
      'SELECT * FROM serialized_episode_memories WHERE series_id = ? AND episode_number = ? LIMIT 1',
      [seriesId, Math.max(1, Number(episodeNumber || 1))]
    );
    return this.parseSerializedEpisodeMemory(row);
  }

  async listSerializedEpisodeMemories(seriesId, limit = 250) {
    const rows = await this.getAllRows(
      `SELECT * FROM serialized_episode_memories
       WHERE series_id = ? ORDER BY episode_number DESC LIMIT ?`,
      [seriesId, Math.max(1, Math.min(2000, Number(limit || 250)))]
    );
    return rows.map(row => this.parseSerializedEpisodeMemory(row));
  }

  async listSerializedEpisodeMemoryRevisions(memoryId, limit = 100) {
    const rows = await this.getAllRows(
      `SELECT * FROM serialized_episode_memory_revisions
       WHERE episode_memory_id = ? ORDER BY revision_number DESC LIMIT ?`,
      [memoryId, Math.max(1, Math.min(1000, Number(limit || 100)))]
    );
    return rows.map(row => {
      let snapshot = {};
      try { snapshot = JSON.parse(row.snapshot || '{}'); } catch (_error) {}
      return {
        id: row.id,
        episodeMemoryId: row.episode_memory_id,
        seriesId: row.series_id,
        episodeNumber: Number(row.episode_number || 0),
        revisionNumber: Number(row.revision_number || 1),
        snapshot,
        changeKind: row.change_kind || 'finalize',
        changeReason: row.change_reason || '',
        actor: row.actor || null,
        createdAt: row.created_at
      };
    });
  }

  async getSerializedEpisodeMemoryCommit(commitId) {
    const row = await this.getRow('SELECT * FROM serialized_episode_memory_commits WHERE id = ? LIMIT 1', [commitId]);
    if (!row) return null;
    let timelineEventIds = [];
    try { timelineEventIds = JSON.parse(row.timeline_event_ids || '[]'); } catch (_error) {}
    return {
      id: row.id,
      seriesId: row.series_id,
      episodeMemoryId: row.episode_memory_id,
      episodeNumber: Number(row.episode_number || 0),
      memoryRevisionNumber: Number(row.memory_revision_number || 1),
      bibleRevisionNumber: Number(row.bible_revision_number || 1),
      commitKind: row.commit_kind || 'finalize',
      timelineEventIds,
      reason: row.reason || '',
      actor: row.actor || null,
      createdAt: row.created_at
    };
  }

  async listSerializedEpisodeMemoryCommits(seriesId, limit = 250) {
    const rows = await this.getAllRows(
      `SELECT * FROM serialized_episode_memory_commits
       WHERE series_id = ? ORDER BY episode_number DESC, created_at DESC LIMIT ?`,
      [seriesId, Math.max(1, Math.min(1000, Number(limit || 250)))]
    );
    const out = [];
    for (const row of rows) out.push(await this.getSerializedEpisodeMemoryCommit(row.id));
    return out;
  }

  async finalizeSerializedEpisodeMemoryAtomic(input = {}) {
    if (!input.seriesId) throw new Error('episode_memory_series_id_required');
    if (!input.memory || !input.memory.episodeNumber) throw new Error('episode_memory_payload_required');

    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const series = await this.getRow('SELECT * FROM serialized_series_bibles WHERE id = ? LIMIT 1', [input.seriesId]);
      if (!series) throw new Error('episode_memory_series_not_found');
      if ((series.status || 'active') !== 'active') throw new Error('episode_memory_series_inactive');

      const currentBibleRevision = Number(series.revision_number || 1);
      const expectedBibleRevision = input.expectedBibleRevision === undefined || input.expectedBibleRevision === null
        ? currentBibleRevision
        : Number(input.expectedBibleRevision);
      if (expectedBibleRevision !== currentBibleRevision) throw new Error('episode_memory_bible_revision_conflict');

      const episodeNumber = Math.max(1, Number(input.memory.episodeNumber || 1));
      const expectedEpisode = Number(series.current_episode || 0) + 1;
      if (episodeNumber !== expectedEpisode) throw new Error('episode_memory_must_finalize_next_episode');

      const duplicate = await this.getRow(
        'SELECT id FROM serialized_episode_memories WHERE series_id = ? AND episode_number = ? LIMIT 1',
        [input.seriesId, episodeNumber]
      );
      if (duplicate) throw new Error('episode_memory_already_finalized');

      const timelineRows = await this.getAllRows(
        `SELECT id FROM serialized_timeline_events
         WHERE series_id = ? AND episode_number = ? AND status = 'active' ORDER BY id ASC`,
        [input.seriesId, episodeNumber]
      );
      const actualTimelineIds = timelineRows.map(row => row.id).sort();
      const providedTimelineIds = [...new Set(input.memory.timelineEventIds || [])].sort();
      if (JSON.stringify(actualTimelineIds) !== JSON.stringify(providedTimelineIds)) {
        throw new Error('episode_memory_timeline_set_mismatch');
      }

      const now = new Date().toISOString();
      const memoryId = input.memory.id || this.generateId('serialized_episode_memory');
      const actor = input.actor || input.memory.approvedBy || null;
      await this.executeQuery(
        `INSERT INTO serialized_episode_memories (
          id, series_id, episode_number, strategy_id, production_id, title, summary,
          discoveries, established_facts, resolved_questions, unresolved_questions,
          narrative_promises, cliffhangers, timeline_event_ids, approval_id, approved_by,
          approval_reason, approved_at, source_fingerprint, revision_number, status,
          created_at, updated_at
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1, 'finalized', ?, ?)`,
        [
          memoryId, input.seriesId, episodeNumber, input.memory.strategyId || null,
          input.memory.productionId || null, input.memory.title || '', input.memory.summary || '',
          JSON.stringify(input.memory.discoveries || []), JSON.stringify(input.memory.establishedFacts || []),
          JSON.stringify(input.memory.resolvedQuestions || []), JSON.stringify(input.memory.unresolvedQuestions || []),
          JSON.stringify(input.memory.narrativePromises || []), JSON.stringify(input.memory.cliffhangers || []),
          JSON.stringify(providedTimelineIds), input.memory.approvalId, input.memory.approvedBy,
          input.memory.approvalReason, input.memory.approvedAt || now, input.memory.sourceFingerprint || null,
          now, now
        ]
      );

      const savedMemory = await this.getSerializedEpisodeMemoryById(memoryId);
      await this.executeQuery(
        `INSERT INTO serialized_episode_memory_revisions (
          id, episode_memory_id, series_id, episode_number, revision_number,
          snapshot, change_kind, change_reason, actor, created_at
        ) VALUES (?, ?, ?, ?, 1, ?, 'finalize', ?, ?, ?)`,
        [
          this.generateId('serialized_episode_memory_revision'), memoryId, input.seriesId, episodeNumber,
          JSON.stringify(savedMemory), input.reason || 'episode_finalize', actor, now
        ]
      );

      const nextBibleRevision = currentBibleRevision + 1;
      await this.executeQuery(
        `UPDATE serialized_series_bibles
         SET current_episode = ?, revision_number = ?, updated_at = ?
         WHERE id = ? AND revision_number = ? AND current_episode = ?`,
        [episodeNumber, nextBibleRevision, now, input.seriesId, currentBibleRevision, Number(series.current_episode || 0)]
      );
      const advanced = await this.getRow('SELECT * FROM serialized_series_bibles WHERE id = ? LIMIT 1', [input.seriesId]);
      if (Number(advanced?.current_episode || -1) !== episodeNumber || Number(advanced?.revision_number || -1) !== nextBibleRevision) {
        throw new Error('episode_memory_bible_revision_conflict');
      }

      const parse = (value, fallback) => { try { return JSON.parse(value || JSON.stringify(fallback)); } catch (_error) { return fallback; } };
      const bibleSnapshot = {
        id: advanced.id,
        namespace: advanced.namespace || 'default',
        seriesKey: advanced.series_key,
        title: advanced.title,
        format: advanced.format || 'serialized',
        genres: parse(advanced.genres, []),
        premise: advanced.premise || '',
        worldRules: parse(advanced.world_rules, []),
        immutableCanon: parse(advanced.immutable_canon, []),
        narrativeRules: parse(advanced.narrative_rules, []),
        centralConflicts: parse(advanced.central_conflicts, []),
        plannedEnding: parse(advanced.planned_ending, null),
        currentEpisode: Number(advanced.current_episode || 0),
        canonVersion: Number(advanced.canon_version || 1),
        revisionNumber: Number(advanced.revision_number || 1),
        status: advanced.status || 'active'
      };
      await this.executeQuery(
        `INSERT INTO serialized_series_bible_revisions (
          id, series_id, revision_number, canon_version, snapshot, change_kind, change_reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, 'episode_finalize', ?, ?, ?)`,
        [
          this.generateId('serialized_series_revision'), input.seriesId, nextBibleRevision,
          Number(advanced.canon_version || 1), JSON.stringify(bibleSnapshot),
          input.reason || `finalize_episode_${episodeNumber}`, actor, now
        ]
      );

      const commitId = input.commitId || this.generateId('serialized_episode_memory_commit');
      await this.executeQuery(
        `INSERT INTO serialized_episode_memory_commits (
          id, series_id, episode_memory_id, episode_number, memory_revision_number,
          bible_revision_number, commit_kind, timeline_event_ids, reason, actor, created_at
        ) VALUES (?, ?, ?, ?, 1, ?, 'finalize', ?, ?, ?, ?)`,
        [
          commitId, input.seriesId, memoryId, episodeNumber, nextBibleRevision,
          JSON.stringify(providedTimelineIds), input.reason || 'episode_finalize', actor, now
        ]
      );

      await this.executeQuery('COMMIT');
      return {
        memory: await this.getSerializedEpisodeMemoryById(memoryId),
        bible: await this.getSerializedSeriesBible(input.seriesId),
        commit: await this.getSerializedEpisodeMemoryCommit(commitId)
      };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }

  async amendSerializedEpisodeMemoryAtomic(input = {}) {
    if (!input.seriesId || !input.memory?.id) throw new Error('episode_memory_identity_required');
    await this.executeQuery('BEGIN IMMEDIATE');
    try {
      const row = await this.getRow(
        'SELECT * FROM serialized_episode_memories WHERE id = ? AND series_id = ? LIMIT 1',
        [input.memory.id, input.seriesId]
      );
      if (!row) throw new Error('episode_memory_not_found');
      const currentRevision = Number(row.revision_number || 1);
      const expectedRevision = Number(input.expectedRevision || currentRevision);
      if (expectedRevision !== currentRevision) throw new Error('episode_memory_revision_conflict');

      const episodeNumber = Number(row.episode_number || 0);
      const timelineRows = await this.getAllRows(
        `SELECT id FROM serialized_timeline_events
         WHERE series_id = ? AND episode_number = ? AND status = 'active' ORDER BY id ASC`,
        [input.seriesId, episodeNumber]
      );
      const actualTimelineIds = timelineRows.map(item => item.id).sort();
      const providedTimelineIds = [...new Set(input.memory.timelineEventIds || [])].sort();
      if (JSON.stringify(actualTimelineIds) !== JSON.stringify(providedTimelineIds)) {
        throw new Error('episode_memory_timeline_set_mismatch');
      }

      const now = new Date().toISOString();
      const nextRevision = currentRevision + 1;
      await this.executeQuery(
        `UPDATE serialized_episode_memories SET
          strategy_id = ?, production_id = ?, title = ?, summary = ?, discoveries = ?,
          established_facts = ?, resolved_questions = ?, unresolved_questions = ?,
          narrative_promises = ?, cliffhangers = ?, timeline_event_ids = ?, source_fingerprint = ?,
          revision_number = ?, updated_at = ?
         WHERE id = ? AND series_id = ? AND revision_number = ?`,
        [
          input.memory.strategyId || null, input.memory.productionId || null,
          input.memory.title || '', input.memory.summary || '', JSON.stringify(input.memory.discoveries || []),
          JSON.stringify(input.memory.establishedFacts || []), JSON.stringify(input.memory.resolvedQuestions || []),
          JSON.stringify(input.memory.unresolvedQuestions || []), JSON.stringify(input.memory.narrativePromises || []),
          JSON.stringify(input.memory.cliffhangers || []), JSON.stringify(providedTimelineIds),
          input.memory.sourceFingerprint || null, nextRevision, now,
          input.memory.id, input.seriesId, currentRevision
        ]
      );
      const confirmed = await this.getRow('SELECT revision_number FROM serialized_episode_memories WHERE id = ? LIMIT 1', [input.memory.id]);
      if (Number(confirmed?.revision_number || -1) !== nextRevision) throw new Error('episode_memory_revision_conflict');

      const saved = await this.getSerializedEpisodeMemoryById(input.memory.id);
      await this.executeQuery(
        `INSERT INTO serialized_episode_memory_revisions (
          id, episode_memory_id, series_id, episode_number, revision_number,
          snapshot, change_kind, change_reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?)`,
        [
          this.generateId('serialized_episode_memory_revision'), saved.id, input.seriesId,
          episodeNumber, nextRevision, JSON.stringify(saved), input.reason || '', input.actor || null, now
        ]
      );
      const series = await this.getRow('SELECT revision_number FROM serialized_series_bibles WHERE id = ? LIMIT 1', [input.seriesId]);
      const commitId = input.commitId || this.generateId('serialized_episode_memory_commit');
      await this.executeQuery(
        `INSERT INTO serialized_episode_memory_commits (
          id, series_id, episode_memory_id, episode_number, memory_revision_number,
          bible_revision_number, commit_kind, timeline_event_ids, reason, actor, created_at
        ) VALUES (?, ?, ?, ?, ?, ?, 'amendment', ?, ?, ?, ?)`,
        [
          commitId, input.seriesId, saved.id, episodeNumber, nextRevision,
          Number(series?.revision_number || 1), JSON.stringify(providedTimelineIds),
          input.reason || '', input.actor || null, now
        ]
      );
      await this.executeQuery('COMMIT');
      return { memory: await this.getSerializedEpisodeMemoryById(saved.id), commit: await this.getSerializedEpisodeMemoryCommit(commitId) };
    } catch (error) {
      await this.executeQuery('ROLLBACK');
      throw error;
    }
  }
