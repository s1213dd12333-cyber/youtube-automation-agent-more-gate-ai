      // Phase 11.12.5 serialized Relationship State Graph
      `CREATE TABLE IF NOT EXISTS serialized_relationship_states (
        id TEXT PRIMARY KEY,
        series_id TEXT NOT NULL,
        source_character_key TEXT NOT NULL,
        target_character_key TEXT NOT NULL,
        source_character_arc_id TEXT NOT NULL,
        target_character_arc_id TEXT NOT NULL,
        relationship_label TEXT NOT NULL DEFAULT '',
        relationship_state TEXT NOT NULL DEFAULT '',
        relationship_tags TEXT NOT NULL DEFAULT '[]',
        trust_score INTEGER NOT NULL DEFAULT 0,
        affinity_score INTEGER NOT NULL DEFAULT 0,
        respect_score INTEGER NOT NULL DEFAULT 0,
        loyalty_score INTEGER NOT NULL DEFAULT 0,
        fear_score INTEGER NOT NULL DEFAULT 0,
        attraction_score INTEGER NOT NULL DEFAULT 0,
        dependence_score INTEGER NOT NULL DEFAULT 0,
        beliefs_about_target TEXT NOT NULL DEFAULT '[]',
        knowledge_about_target TEXT NOT NULL DEFAULT '[]',
        secrets_known_about_target TEXT NOT NULL DEFAULT '[]',
        obligations_to_target TEXT NOT NULL DEFAULT '[]',
        promises_to_target TEXT NOT NULL DEFAULT '[]',
        grievances_against_target TEXT NOT NULL DEFAULT '[]',
        expectations_of_target TEXT NOT NULL DEFAULT '[]',
        boundaries_with_target TEXT NOT NULL DEFAULT '[]',
        shared_history TEXT NOT NULL DEFAULT '[]',
        current_tensions TEXT NOT NULL DEFAULT '[]',
        notes TEXT NOT NULL DEFAULT '',
        first_episode_number INTEGER NOT NULL DEFAULT 0,
        last_episode_number INTEGER NOT NULL DEFAULT 0,
        revision_number INTEGER NOT NULL DEFAULT 0,
        status TEXT NOT NULL DEFAULT 'active',
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        UNIQUE(series_id, source_character_key, target_character_key),
        CHECK(source_character_key <> target_character_key),
        CHECK(trust_score BETWEEN -100 AND 100),
        CHECK(affinity_score BETWEEN -100 AND 100),
        CHECK(respect_score BETWEEN -100 AND 100),
        CHECK(loyalty_score BETWEEN -100 AND 100),
        CHECK(fear_score BETWEEN 0 AND 100),
        CHECK(attraction_score BETWEEN 0 AND 100),
        CHECK(dependence_score BETWEEN 0 AND 100),
        FOREIGN KEY (series_id) REFERENCES serialized_series_bibles(id) ON DELETE CASCADE,
        FOREIGN KEY (source_character_arc_id) REFERENCES serialized_character_arcs(id) ON DELETE CASCADE,
        FOREIGN KEY (target_character_arc_id) REFERENCES serialized_character_arcs(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_relationship_series ON serialized_relationship_states(series_id, status, last_episode_number DESC)`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_relationship_source ON serialized_relationship_states(series_id, source_character_key, status)`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_relationship_target ON serialized_relationship_states(series_id, target_character_key, status)`,
      `CREATE TABLE IF NOT EXISTS serialized_relationship_revisions (
        id TEXT PRIMARY KEY,
        relationship_id TEXT NOT NULL,
        series_id TEXT NOT NULL,
        episode_number INTEGER NOT NULL,
        revision_number INTEGER NOT NULL,
        snapshot TEXT NOT NULL DEFAULT '{}',
        change_kind TEXT NOT NULL DEFAULT 'episode_commit',
        change_reason TEXT NOT NULL DEFAULT '',
        actor TEXT,
        created_at TEXT NOT NULL,
        UNIQUE(relationship_id, revision_number),
        FOREIGN KEY (relationship_id) REFERENCES serialized_relationship_states(id) ON DELETE CASCADE,
        FOREIGN KEY (series_id) REFERENCES serialized_series_bibles(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_relationship_revisions_edge ON serialized_relationship_revisions(relationship_id, revision_number DESC)`,
      `CREATE TABLE IF NOT EXISTS serialized_relationship_episode_commits (
        id TEXT PRIMARY KEY,
        series_id TEXT NOT NULL,
        relationship_id TEXT NOT NULL,
        episode_memory_id TEXT NOT NULL,
        episode_number INTEGER NOT NULL,
        relationship_revision_number INTEGER NOT NULL,
        commit_kind TEXT NOT NULL DEFAULT 'episode_commit',
        timeline_event_ids TEXT NOT NULL DEFAULT '[]',
        before_snapshot TEXT NOT NULL DEFAULT '{}',
        after_snapshot TEXT NOT NULL DEFAULT '{}',
        change_summary TEXT NOT NULL DEFAULT '',
        reason TEXT NOT NULL DEFAULT '',
        actor TEXT,
        created_at TEXT NOT NULL,
        FOREIGN KEY (series_id) REFERENCES serialized_series_bibles(id) ON DELETE CASCADE,
        FOREIGN KEY (relationship_id) REFERENCES serialized_relationship_states(id) ON DELETE CASCADE,
        FOREIGN KEY (episode_memory_id) REFERENCES serialized_episode_memories(id) ON DELETE CASCADE
      )`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_relationship_commits_edge ON serialized_relationship_episode_commits(relationship_id, episode_number DESC, relationship_revision_number DESC)`,
      `CREATE INDEX IF NOT EXISTS idx_serialized_relationship_commits_series ON serialized_relationship_episode_commits(series_id, episode_number DESC, created_at DESC)`,
