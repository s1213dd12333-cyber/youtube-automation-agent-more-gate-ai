  async createSerializedSeriesBible(bible = {}) {
    const now = new Date().toISOString();
    const id = bible.id || this.generateId('serialized_series');
    await this.executeQuery(
      `INSERT OR IGNORE INTO serialized_series_bibles (
        id, namespace, series_key, title, format, genres, premise, world_rules, immutable_canon,
        narrative_rules, central_conflicts, planned_ending, current_episode, canon_version,
        revision_number, status, created_by, created_at, updated_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        id, bible.namespace || 'default', bible.seriesKey || 'series', bible.title || 'Untitled Series',
        bible.format || 'serialized', JSON.stringify(bible.genres || []), bible.premise || '',
        JSON.stringify(bible.worldRules || []), JSON.stringify(bible.immutableCanon || []),
        JSON.stringify(bible.narrativeRules || []), JSON.stringify(bible.centralConflicts || []),
        JSON.stringify(bible.plannedEnding ?? null), Math.max(0, Number(bible.currentEpisode || 0)),
        Math.max(1, Number(bible.canonVersion || 1)), Math.max(1, Number(bible.revisionNumber || 1)),
        bible.status || 'active', bible.createdBy || null, bible.createdAt || now, now
      ]
    );
    return this.getSerializedSeriesBibleByKey(bible.namespace || 'default', bible.seriesKey || 'series');
  }

  async updateSerializedSeriesBible(bible = {}, expectedRevisionNumber) {
    const now = new Date().toISOString();
    const expected = Math.max(1, Number(expectedRevisionNumber || 1));
    await this.executeQuery(
      `UPDATE serialized_series_bibles SET
        title = ?, format = ?, genres = ?, premise = ?, world_rules = ?, immutable_canon = ?,
        narrative_rules = ?, central_conflicts = ?, planned_ending = ?, current_episode = ?,
        canon_version = ?, revision_number = ?, status = ?, updated_at = ?
       WHERE id = ? AND revision_number = ?`,
      [
        bible.title || 'Untitled Series', bible.format || 'serialized', JSON.stringify(bible.genres || []),
        bible.premise || '', JSON.stringify(bible.worldRules || []), JSON.stringify(bible.immutableCanon || []),
        JSON.stringify(bible.narrativeRules || []), JSON.stringify(bible.centralConflicts || []),
        JSON.stringify(bible.plannedEnding ?? null), Math.max(0, Number(bible.currentEpisode || 0)),
        Math.max(1, Number(bible.canonVersion || 1)), Math.max(1, Number(bible.revisionNumber || expected + 1)),
        bible.status || 'active', now, bible.id, expected
      ]
    );
    const saved = await this.getSerializedSeriesBible(bible.id);
    return saved && Number(saved.revisionNumber) === Number(bible.revisionNumber) ? saved : null;
  }

  async getSerializedSeriesBible(seriesId) {
    const row = await this.getRow('SELECT * FROM serialized_series_bibles WHERE id = ? LIMIT 1', [seriesId]);
    return this.parseSerializedSeriesBible(row);
  }

  async getSerializedSeriesBibleByKey(namespace, seriesKey) {
    const row = await this.getRow(
      'SELECT * FROM serialized_series_bibles WHERE namespace = ? AND series_key = ? LIMIT 1',
      [namespace || 'default', seriesKey || '']
    );
    return this.parseSerializedSeriesBible(row);
  }

  async listSerializedSeriesBibles(namespace = 'default', limit = 250) {
    const rows = await this.getAllRows(
      'SELECT * FROM serialized_series_bibles WHERE namespace = ? ORDER BY updated_at DESC, rowid DESC LIMIT ?',
      [namespace || 'default', Math.max(1, Math.min(1000, Number(limit || 250)))]
    );
    return rows.map(row => this.parseSerializedSeriesBible(row));
  }

  async saveSerializedSeriesBibleRevision(revision = {}) {
    const now = new Date().toISOString();
    const id = revision.id || this.generateId('serialized_series_revision');
    await this.executeQuery(
      `INSERT OR IGNORE INTO serialized_series_bible_revisions (
        id, series_id, revision_number, canon_version, snapshot, change_kind, change_reason, actor, created_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        id, revision.seriesId, Math.max(1, Number(revision.revisionNumber || 1)),
        Math.max(1, Number(revision.canonVersion || 1)), JSON.stringify(revision.snapshot || {}),
        revision.changeKind || 'update', revision.changeReason || '', revision.actor || null, revision.createdAt || now
      ]
    );
    const row = await this.getRow(
      'SELECT * FROM serialized_series_bible_revisions WHERE series_id = ? AND revision_number = ? LIMIT 1',
      [revision.seriesId, Math.max(1, Number(revision.revisionNumber || 1))]
    );
    return this.parseSerializedSeriesBibleRevision(row);
  }

  async listSerializedSeriesBibleRevisions(seriesId, limit = 100) {
    const rows = await this.getAllRows(
      'SELECT * FROM serialized_series_bible_revisions WHERE series_id = ? ORDER BY revision_number DESC LIMIT ?',
      [seriesId, Math.max(1, Math.min(1000, Number(limit || 100)))]
    );
    return rows.map(row => this.parseSerializedSeriesBibleRevision(row));
  }

  async bindSerializedSeriesStrategy(binding = {}) {
    const now = new Date().toISOString();
    await this.executeQuery(
      `INSERT INTO serialized_series_strategy_bindings (
        strategy_id, series_id, episode_number, actor, reason, created_at, updated_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(strategy_id) DO UPDATE SET
        series_id = excluded.series_id,
        episode_number = excluded.episode_number,
        actor = excluded.actor,
        reason = excluded.reason,
        updated_at = excluded.updated_at`,
      [
        binding.strategyId, binding.seriesId, Math.max(1, Number(binding.episodeNumber || 1)),
        binding.actor || null, binding.reason || '', binding.createdAt || now, now
      ]
    );
    return this.getSerializedSeriesStrategyBinding(binding.strategyId);
  }

  async getSerializedSeriesStrategyBinding(strategyId) {
    const row = await this.getRow('SELECT * FROM serialized_series_strategy_bindings WHERE strategy_id = ? LIMIT 1', [strategyId]);
    return this.parseSerializedSeriesStrategyBinding(row);
  }

  async listSerializedSeriesStrategyBindings(seriesId, limit = 250) {
    const rows = await this.getAllRows(
      'SELECT * FROM serialized_series_strategy_bindings WHERE series_id = ? ORDER BY episode_number ASC, created_at ASC LIMIT ?',
      [seriesId, Math.max(1, Math.min(1000, Number(limit || 250)))]
    );
    return rows.map(row => this.parseSerializedSeriesStrategyBinding(row));
  }

  parseSerializedSeriesBible(row) {
    if (!row) return null;
    const parse = (value, fallback) => {
      try { return JSON.parse(value); } catch (_error) { return fallback; }
    };
    return {
      id: row.id,
      namespace: row.namespace || 'default',
      seriesKey: row.series_key,
      title: row.title,
      format: row.format || 'serialized',
      genres: parse(row.genres || '[]', []),
      premise: row.premise || '',
      worldRules: parse(row.world_rules || '[]', []),
      immutableCanon: parse(row.immutable_canon || '[]', []),
      narrativeRules: parse(row.narrative_rules || '[]', []),
      centralConflicts: parse(row.central_conflicts || '[]', []),
      plannedEnding: parse(row.planned_ending || 'null', null),
      currentEpisode: Number(row.current_episode || 0),
      canonVersion: Number(row.canon_version || 1),
      revisionNumber: Number(row.revision_number || 1),
      status: row.status || 'active',
      createdBy: row.created_by || null,
      createdAt: row.created_at,
      updatedAt: row.updated_at
    };
  }

  parseSerializedSeriesBibleRevision(row) {
    if (!row) return null;
    let snapshot = {};
    try { snapshot = JSON.parse(row.snapshot || '{}'); } catch (_error) {}
    return {
      id: row.id,
      seriesId: row.series_id,
      revisionNumber: Number(row.revision_number || 1),
      canonVersion: Number(row.canon_version || 1),
      snapshot,
      changeKind: row.change_kind || 'update',
      changeReason: row.change_reason || '',
      actor: row.actor || null,
      createdAt: row.created_at
    };
  }

  parseSerializedSeriesStrategyBinding(row) {
    if (!row) return null;
    return {
      strategyId: row.strategy_id,
      seriesId: row.series_id,
      episodeNumber: Math.max(1, Number(row.episode_number || 1)),
      actor: row.actor || null,
      reason: row.reason || '',
      createdAt: row.created_at,
      updatedAt: row.updated_at
    };
  }
