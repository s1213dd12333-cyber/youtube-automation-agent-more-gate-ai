  async savePersistentWorldObjectAlias(input = {}) {
    if (!input.objectId || !input.aliasKey) return null;
    const namespace = input.namespace || 'default';
    const existing = await this.getRow('SELECT id, created_at FROM persistent_world_object_aliases WHERE namespace = ? AND object_id = ? AND alias_key = ? LIMIT 1', [namespace, input.objectId, input.aliasKey]);
    const id = existing?.id || input.id || this.generateId('persistent_world_object_alias');
    const now = new Date().toISOString(); const createdAt = existing?.created_at || input.createdAt || now;
    await this.executeQuery(`INSERT INTO persistent_world_object_aliases (id, namespace, object_id, alias_text, alias_key, canonical, source_kind, created_at, updated_at)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(namespace, object_id, alias_key) DO UPDATE SET alias_text=excluded.alias_text,
      canonical=MAX(persistent_world_object_aliases.canonical, excluded.canonical), source_kind=excluded.source_kind, updated_at=excluded.updated_at`,
      [id, namespace, input.objectId, input.aliasText || input.aliasKey, input.aliasKey, input.canonical ? 1 : 0, input.sourceKind || 'object_registration', createdAt, now]);
    return this.parsePersistentWorldObjectAlias(await this.getRow('SELECT * FROM persistent_world_object_aliases WHERE id = ? LIMIT 1', [id]));
  }

  async listPersistentWorldObjectAliases(objectId) {
    const rows = await this.getAllRows('SELECT * FROM persistent_world_object_aliases WHERE object_id = ? ORDER BY canonical DESC, created_at, rowid', [objectId]);
    return rows.map(row => this.parsePersistentWorldObjectAlias(row));
  }

  async listPersistentWorldObjectAliasesByKey(namespace, aliasKey) {
    const rows = await this.getAllRows('SELECT * FROM persistent_world_object_aliases WHERE namespace = ? AND alias_key = ? ORDER BY canonical DESC, created_at, rowid', [namespace || 'default', aliasKey]);
    return rows.map(row => this.parsePersistentWorldObjectAlias(row));
  }

  async savePersistentWorldObjectResolution(input = {}) {
    if (!input.productionId || !input.scopeKey || input.referenceKey == null || !input.status) return null;
    const namespace = input.namespace || 'default';
    const existing = await this.getRow('SELECT id, created_at FROM persistent_world_object_resolutions WHERE production_id = ? AND scope_key = ? AND reference_key = ? LIMIT 1', [input.productionId, input.scopeKey, input.referenceKey]);
    const id = existing?.id || input.id || this.generateId('persistent_world_object_resolution');
    const now = new Date().toISOString(); const createdAt = existing?.created_at || input.createdAt || now;
    await this.executeQuery(`INSERT INTO persistent_world_object_resolutions (id, namespace, production_id, scope_key, reference_text, reference_key, requested_type, status, object_id, match_mode, confidence, candidate_object_ids, reason, identity_fingerprint, created_at, updated_at)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(production_id, scope_key, reference_key) DO UPDATE SET
      namespace=excluded.namespace, reference_text=excluded.reference_text, requested_type=excluded.requested_type, status=excluded.status, object_id=excluded.object_id,
      match_mode=excluded.match_mode, confidence=excluded.confidence, candidate_object_ids=excluded.candidate_object_ids, reason=excluded.reason,
      identity_fingerprint=excluded.identity_fingerprint, updated_at=excluded.updated_at`,
      [id, namespace, input.productionId, input.scopeKey, input.referenceText || '', input.referenceKey || '', input.requestedType || null, input.status, input.objectId || null,
       input.matchMode || null, Number(input.confidence || 0), JSON.stringify(input.candidateObjectIds || []), input.reason || null, input.identityFingerprint || null, createdAt, now]);
    return this.parsePersistentWorldObjectResolution(await this.getRow('SELECT * FROM persistent_world_object_resolutions WHERE id = ? LIMIT 1', [id]));
  }

  async listPersistentWorldObjectResolutions(productionId) {
    const rows = await this.getAllRows('SELECT * FROM persistent_world_object_resolutions WHERE production_id = ? ORDER BY updated_at, rowid', [productionId]);
    return rows.map(row => this.parsePersistentWorldObjectResolution(row));
  }

  parsePersistentWorldObjectAlias(row) {
    if (!row) return null;
    return { id: row.id, namespace: row.namespace || 'default', objectId: row.object_id, aliasText: row.alias_text, aliasKey: row.alias_key,
      canonical: Number(row.canonical || 0) === 1, sourceKind: row.source_kind || 'object_registration', createdAt: row.created_at, updatedAt: row.updated_at };
  }

  parsePersistentWorldObjectResolution(row) {
    if (!row) return null;
    return { id: row.id, namespace: row.namespace || 'default', productionId: row.production_id, scopeKey: row.scope_key, referenceText: row.reference_text,
      referenceKey: row.reference_key, requestedType: row.requested_type, status: row.status, objectId: row.object_id, matchMode: row.match_mode,
      confidence: Number(row.confidence || 0), candidateObjectIds: JSON.parse(row.candidate_object_ids || '[]'), reason: row.reason,
      identityFingerprint: row.identity_fingerprint, createdAt: row.created_at, updatedAt: row.updated_at };
  }
