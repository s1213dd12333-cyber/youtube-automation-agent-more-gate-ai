  async savePersistentCharacterAlias(input = {}) {
    if (!input.characterId || !input.aliasKey) return null;
    const namespace = input.namespace || 'default';
    const existing = await this.getRow('SELECT id, created_at FROM persistent_character_aliases WHERE namespace = ? AND character_id = ? AND alias_key = ? LIMIT 1', [namespace, input.characterId, input.aliasKey]);
    const id = existing?.id || input.id || this.generateId('persistent_character_alias');
    const now = new Date().toISOString(); const createdAt = existing?.created_at || input.createdAt || now;
    await this.executeQuery(`INSERT INTO persistent_character_aliases (id, namespace, character_id, alias_text, alias_key, canonical, source_kind, created_at, updated_at)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(namespace, character_id, alias_key) DO UPDATE SET alias_text=excluded.alias_text,
      canonical=MAX(persistent_character_aliases.canonical, excluded.canonical), source_kind=excluded.source_kind, updated_at=excluded.updated_at`,
      [id, namespace, input.characterId, input.aliasText || input.aliasKey, input.aliasKey, input.canonical ? 1 : 0, input.sourceKind || 'character_registration', createdAt, now]);
    return this.parsePersistentCharacterAlias(await this.getRow('SELECT * FROM persistent_character_aliases WHERE id = ? LIMIT 1', [id]));
  }

  async listPersistentCharacterAliases(characterId) {
    const rows = await this.getAllRows('SELECT * FROM persistent_character_aliases WHERE character_id = ? ORDER BY canonical DESC, created_at, rowid', [characterId]);
    return rows.map(row => this.parsePersistentCharacterAlias(row));
  }

  async listPersistentCharacterAliasesByKey(namespace, aliasKey) {
    const rows = await this.getAllRows('SELECT * FROM persistent_character_aliases WHERE namespace = ? AND alias_key = ? ORDER BY canonical DESC, created_at, rowid', [namespace || 'default', aliasKey]);
    return rows.map(row => this.parsePersistentCharacterAlias(row));
  }

  async savePersistentCharacterResolution(input = {}) {
    if (!input.productionId || !input.scopeKey || input.referenceKey == null || !input.status) return null;
    const namespace = input.namespace || 'default';
    const existing = await this.getRow('SELECT id, created_at FROM persistent_character_resolutions WHERE production_id = ? AND scope_key = ? AND reference_key = ? LIMIT 1', [input.productionId, input.scopeKey, input.referenceKey]);
    const id = existing?.id || input.id || this.generateId('persistent_character_resolution');
    const now = new Date().toISOString(); const createdAt = existing?.created_at || input.createdAt || now;
    await this.executeQuery(`INSERT INTO persistent_character_resolutions (id, namespace, production_id, scope_key, reference_text, reference_key, requested_species, status, character_id, match_mode, confidence, candidate_character_ids, reason, identity_fingerprint, created_at, updated_at)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(production_id, scope_key, reference_key) DO UPDATE SET
      namespace=excluded.namespace, reference_text=excluded.reference_text, requested_species=excluded.requested_species, status=excluded.status, character_id=excluded.character_id,
      match_mode=excluded.match_mode, confidence=excluded.confidence, candidate_character_ids=excluded.candidate_character_ids, reason=excluded.reason,
      identity_fingerprint=excluded.identity_fingerprint, updated_at=excluded.updated_at`,
      [id, namespace, input.productionId, input.scopeKey, input.referenceText || '', input.referenceKey || '', input.requestedSpecies || null, input.status, input.characterId || null,
       input.matchMode || null, Number(input.confidence || 0), JSON.stringify(input.candidateCharacterIds || []), input.reason || null, input.identityFingerprint || null, createdAt, now]);
    return this.parsePersistentCharacterResolution(await this.getRow('SELECT * FROM persistent_character_resolutions WHERE id = ? LIMIT 1', [id]));
  }

  async listPersistentCharacterResolutions(productionId) {
    const rows = await this.getAllRows('SELECT * FROM persistent_character_resolutions WHERE production_id = ? ORDER BY updated_at, rowid', [productionId]);
    return rows.map(row => this.parsePersistentCharacterResolution(row));
  }

  parsePersistentCharacterAlias(row) {
    if (!row) return null;
    return { id: row.id, namespace: row.namespace || 'default', characterId: row.character_id, aliasText: row.alias_text, aliasKey: row.alias_key,
      canonical: Number(row.canonical || 0) === 1, sourceKind: row.source_kind || 'character_registration', createdAt: row.created_at, updatedAt: row.updated_at };
  }

  parsePersistentCharacterResolution(row) {
    if (!row) return null;
    return { id: row.id, namespace: row.namespace || 'default', productionId: row.production_id, scopeKey: row.scope_key, referenceText: row.reference_text,
      referenceKey: row.reference_key, requestedSpecies: row.requested_species, status: row.status, characterId: row.character_id, matchMode: row.match_mode,
      confidence: Number(row.confidence || 0), candidateCharacterIds: JSON.parse(row.candidate_character_ids || '[]'), reason: row.reason,
      identityFingerprint: row.identity_fingerprint, createdAt: row.created_at, updatedAt: row.updated_at };
  }
